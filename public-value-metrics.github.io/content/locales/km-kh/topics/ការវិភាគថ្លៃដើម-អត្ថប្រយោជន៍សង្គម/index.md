# ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម

ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម (Social Cost-Benefit Analysis, SCBA) បម្លែងរាល់ថ្លៃចំណាយ និងអត្ថប្រយោជន៍នៃគោលនយោបាយ ឬកម្មវិធីមួយ ទាំងដែលមានទីផ្សារ និងគ្មានទីផ្សារ ទៅជាឯកតាប្រាក់រួមតែមួយ បញ្ចុះតម្លៃលំហូរអនាគតមកជាតម្លៃបច្ចុប្បន្ន ហើយកាត់ដកគ្នាដើម្បីបង្កើតលេខតែមួយ៖ តើសំណើនេះធ្វើឱ្យសង្គមប្រសើរឡើងឬទេ ហើយប្រសើរប៉ុន្មាន?

## ហេតុអ្វីបានជាសំខាន់

SCBA គឺជាវិធីសាស្ត្របរិមាណលំនាំដើមនៅក្នុងករណីសេដ្ឋកិច្ចនៃ[ការវាយតម្លៃតាម Green Book](../ការវាយតម្លៃតាម-green-book/)៖ ការណែនាំរបស់ក្រសួងរតនាគារអង់គ្លេស (HM Treasury) តម្រូវឱ្យសំណើបង្ហាញតម្លៃសង្គមបច្ចុប្បន្នសុទ្ធ (net present social value, NPSV) ជាវិជ្ជមាន នៅពេលណាដែលអត្ថប្រយោជន៍អាចបម្លែងជាប្រាក់បានដោយគួរឱ្យជឿ ដោយប្រើឆន្ទៈក្នុងការបង់ (willingness-to-pay) ជាគោលការណ៍វាយតម្លៃមូលដ្ឋានសម្រាប់ទំនិញគ្មានទីផ្សារ
(<https://www.gov.uk/government/publications/the-green-book-appraisal-and-evaluation-in-central-government>,
Chapter 5)។ វិន័យដែលវាបង្កើតគឺថា ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍ "សង្គម" មិនដូចគ្នាទៅនឹងការវាយតម្លៃវិនិយោគវិស័យឯកជនទេ៖ វាត្រូវតែរួមបញ្ចូលថ្លៃចំណាយ និងអត្ថប្រយោជន៍ដែលធ្លាក់លើភាគីទីបីដែលមិនមែនជាភាគីនៃប្រតិបត្តិការ (ផលប៉ះពាល់ខាងក្រៅ) វាត្រូវប្រើ[អត្រាបញ្ចុះតម្លៃសង្គម](../អត្រាបញ្ចុះតម្លៃសង្គម/) ជំនួសឱ្យថ្លៃចំណាយដើមទុនពាណិជ្ជកម្ម ហើយវាគួរតែអនុវត្ត[ការថ្លឹងថ្លែងតាមការបែងចែក](../ការថ្លឹងថ្លែងតាមការបែងចែក/) នៅពេលដែលមួយផោនមានសារៈសំខាន់ជាងចំពោះគ្រួសារក្រីក្រជាងគ្រួសារមាន។

SCBA ដួលរលំនៅកន្លែងដែលអ្នករិះគន់រំពឹងទុកយ៉ាងត្រឹមត្រូវ៖ ទំនិញដែលគ្មានភាពស្រដៀងទីផ្សារ ដូចជា ខ្យល់បរិសុទ្ធ សាមគ្គីភាពសង្គម តម្លៃនៃជីវិតមួយដែលបានសង្គ្រោះ ត្រូវតែបម្លែងជាប្រាក់ដោយប្រើវិធីសាស្ត្រ[ចំណូលចិត្តដែលបានប្រកាស](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស/) ឬ[ចំណូលចិត្តដែលបានបង្ហាញ](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានបង្ហាញ/) ឬត្រូវបង្កើត[តម្លៃស្រមោល](../តម្លៃស្រមោល/)។ នៅពេលការបម្លែងជាប្រាក់មានភាពចម្រូងចម្រាស មិនមែនគ្រាន់តែពិបាក នោះ Green Book ផ្ទាល់ណែនាំឱ្យថយក្រោយទៅរក[ការវិភាគប្រសិទ្ធភាពថ្លៃចំណាយ](../ការវិភាគប្រសិទ្ធភាពថ្លៃចំណាយក្នុងរដ្ឋាភិបាល/) ឬ[ការវិភាគការសម្រេចចិត្តតាមលក្ខខណ្ឌច្រើន](../ការវិភាគការសម្រេចចិត្តពហុលក្ខណៈវិនិច្ឆ័យ/) ជាជាងបង្ខំលេខដែលគ្មាននរណាជឿ។

## គណិតវិទ្យា

```
NPSV = Σ [t=0 ដល់ T] (Benefit_t − Cost_t) / (1 + r)^t

ដែល៖
  Benefit_t = អត្ថប្រយោជន៍ដែលបានបម្លែងជាប្រាក់ទាំងអស់ក្នុងឆ្នាំ t រួមទាំងទំនិញ
              គ្មានទីផ្សារដែលត្រូវបានវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស/បានបង្ហាញ ឬតម្លៃស្រមោល
  Cost_t    = ថ្លៃចំណាយដែលបានបម្លែងជាប្រាក់ទាំងអស់ក្នុងឆ្នាំ t រួមទាំងថ្លៃចំណាយឱកាស
              នៃធនធាន (សូមមើល ../opportunity-cost-in-public-spending/)
  r         = អត្រាបញ្ចុះតម្លៃសង្គម (HM Treasury កំណត់ 3.5% ថយចុះទៅ
              អត្រាទាបជាងនេះក្រោយឆ្នាំទី 30 តាមឧបសម្ព័ន្ធ A របស់ Green Book)
  T         = រយៈពេលវាយតម្លៃ

សមាមាត្រអត្ថប្រយោជន៍-ថ្លៃចំណាយ (BCR) = Σ PV(Benefits) / Σ PV(Costs)
```

BCR លើសពី 1 (ឬ NPSV លើសពីសូន្យ) បង្ហាញពីតម្លៃសង្គមសុទ្ធ។ ប្រភេទតម្លៃសមស្របនឹងថ្លៃចំណាយ (value for money) របស់ Green Book (ដូចដែលប្រើក្នុងការវាយតម្លៃដឹកជញ្ជូន និងហេដ្ឋារចនាសម្ព័ន្ធ) កំណត់ជួរ BCR ដូចតទៅ៖ ក្រោម 1.0 គឺតម្លៃសមស្របខ្សោយ 1.0–1.5 គឺទាប 1.5–2.0 គឺមធ្យម 2.0–4.0 គឺខ្ពស់ ហើយលើស 4.0 គឺខ្ពស់ខ្លាំង។ ការវិភាគភាពរសើប (sensitivity analysis) គឺការគណនា NPSV ឡើងវិញក្រោមការសន្មត់ទុទិដ្ឋិនិយម និងសុទិដ្ឋិនិយម ហើយជាកាតព្វកិច្ច មិនមែនជាជម្រើសទេ ពីព្រោះអត្ថប្រយោជន៍គ្មានទីផ្សារដែលបានបម្លែងជាប្រាក់មានជួរមិនប្រាកដប្រជាធំទូលាយ។

## ឧទាហរណ៍គណនា

**អាជ្ញាធរមូលដ្ឋាន**៖ ក្រុមប្រឹក្សាមួយវាយតម្លៃការវិនិយោគ £3m លើបណ្តាញជិះកង់ និងដើរថ្មី ក្នុងរយៈពេលវាយតម្លៃ 20 ឆ្នាំ ដោយអត្រាបញ្ចុះតម្លៃ 3.5%។

```
ថ្លៃចំណាយ៖ ដើមទុន £3m ក្នុងឆ្នាំទី 0, £50,000/ឆ្នាំ សម្រាប់ថែទាំ (ឆ្នាំទី 1-20)
PV(ថែទាំ) ≈ £50,000 × 14.2 (កត្តាប្រាក់បៀវត្សរ៍ 20 ឆ្នាំ ក្នុងអត្រា 3.5%) ≈ £710,000
PV(ថ្លៃចំណាយ) សរុប ≈ £3.71m

អត្ថប្រយោជន៍ (បម្លែងជាប្រាក់ទាំងអស់តាមឧបករណ៍វាយតម្លៃដែលបានផ្សព្វផ្សាយរបស់ DfT/WHO)៖
  អត្ថប្រយោជន៍សុខភាពពីសកម្មភាពរាងកាយកើនឡើង៖ £180,000/ឆ្នាំ
  ការកាត់បន្ថយអវត្តមាន៖ £40,000/ឆ្នាំ
  ការកាត់បន្ថយការកកស្ទះ (ការធ្វើដំណើរដោយឡានតិចជាងមុន)៖ £60,000/ឆ្នាំ
  លំហូរអត្ថប្រយោជន៍សរុប៖ £280,000/ឆ្នាំ
PV(អត្ថប្រយោជន៍) ≈ £280,000 × 14.2 ≈ £3.98m

NPSV = £3.98m − £3.71m = +£0.27m
BCR = 3.98 / 3.71 = 1.07 → តម្លៃសមស្រប "ទាប"
```

គម្រោងនេះឆ្លងកាត់ខ្សែបន្ទាត់ ប៉ុន្តែគ្រាន់តែបន្តិច។ ការគណនាភាពរសើបដោយប៉ាន់ស្មានអត្ថប្រយោជន៍សុខភាពទាបជាង 20% (ឆ្លុះបញ្ចាំងពីភាពមិនប្រាកដប្រជាពិតប្រាកដក្នុងការវាយតម្លៃសកម្មភាពរាងកាយ) ធ្វើឱ្យ BCR ធ្លាក់ក្រោម 1.0 ដែលជាមូលហេតុពិតប្រាកដដែល Green Book តម្រូវឱ្យផ្សព្វផ្សាយតារាងភាពរសើបជាមួយលេខចម្បង មិនមែនគ្រាន់តែការប៉ាន់ស្មានកណ្តាលទេ។

**អង្គការសប្បុរសធម៌**៖ កម្មវិធីការពារមរណភាពទារក ដែលមានថ្លៃចំណាយ £500,000/ឆ្នាំ ត្រូវបានវាយតម្លៃដោយប្រើតម្លៃនៃជីវិតស្ថិតិ (value of a statistical life, VSL) ដែលជាតម្លៃស្រមោល មិនមែនតម្លៃទីផ្សារដែលបានសង្កេតទេ ប្រហែល £2.1m (តួលេខដែលបានធ្វើបច្ចុប្បន្នភាពឆ្នាំ 2023 របស់ HM Treasury ដែលខ្លួនវាបានមកពីការសិក្សាចំណូលចិត្តដែលបានប្រកាស)។ ការជៀសវាងមរណភាពទារកមួយក្នុងមួយឆ្នាំ ធៀបនឹងថ្លៃចំណាយ £500,000 ផ្តល់ BCR 4.2 ដែលជាតម្លៃសមស្រប "ខ្ពស់ខ្លាំង" យ៉ាងងាយស្រួល ប៉ុន្តែលទ្ធផលទាំងមូលអាស្រ័យលើតួលេខ VSL ដែលជាមូលហេតុដែល SCBA ណាមួយដែលប្រើ VSL ត្រូវតែបង្ហាញវាជាការសន្មត់ មិនមែនជាអង្គហេតុទេ។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

SCBA គឺជាក្របខ័ណ្ឌធម្មជាតិសម្រាប់ការសម្រេចចិត្តវិនិយោគលើវេទិកា និងហេដ្ឋារចនាសម្ព័ន្ធក្នុងកម្មវិធីរបស់រដ្ឋាភិបាល ឧទាហរណ៍ ការប្រៀបធៀបវេទិកាអត្តសញ្ញាណរួមមួយ ជាមួយដំណោះស្រាយចំណុចៗរបស់ក្រសួង តម្រូវឱ្យបម្លែងអត្ថប្រយោជន៍ជាប្រាក់ ដូចជា ថ្លៃចំណាយចុះឈ្មោះស្ទួនតិចជាងមុន ការក្លែងបន្លំតិចជាងមុន និងពេលវេលាទៅដល់សេវាលឿនជាងមុន ដែលមិនមានតម្លៃទីផ្សារដោយខ្លួនឯង។ វិស្វករដែលកសាងសេវាមូលដ្ឋានគួររំពឹងថាអ្នកដឹកនាំកម្មវិធីនឹងសុំធាតុបញ្ចូលសម្រាប់ការវិភាគនេះ៖ ថ្លៃចំណាយឯកតានៃប្រតិបត្តិការ (សូមមើល [ថ្លៃចំណាយក្នុងមួយប្រតិបត្តិការ](../ថ្លៃចំណាយក្នុងមួយប្រតិបត្តិការ/)) បរិមាណដែលរំពឹងទុក និងថ្លៃចំណាយនៃការថយចុះគុណភាព/ពេលវេលាដែលប្រព័ន្ធដួល។ វិន័យដែលសំខាន់បំផុតក្នុងការនាំយក គឺបញ្ចុះតម្លៃអត្ថប្រយោជន៍អនាគត ដាក់ឈ្មោះស្ថានភាពមូលដ្ឋានប្រៀបធៀប (counterfactual) ឱ្យច្បាស់លាស់ (សូមមើល [ការវិភាគ counterfactual](../ការវិភាគ-counterfactual/)) ហើយកុំបង្ហាញការប៉ាន់ស្មានចំណុចតែមួយ ដោយគ្មានជួរភាពរសើបរបស់វា។

## កំហុសដែលតែងកើតមាន

- **ការរាប់អត្ថប្រយោជន៍ស្ទួន។** ការរាប់ទាំង "ពេលវេលាដែលបានសន្សំ" និង "ផលិតភាពដែលទទួលបានពីពេលវេលានោះ" ជាបន្ទាត់អត្ថប្រយោជន៍ដាច់ដោយឡែក ធ្វើឱ្យករណីហួសការពិត។ ពេលវេលាដែលបានសន្សំគឺជាអត្ថប្រយោជន៍ ការប្រើប្រាស់បន្តបន្ទាប់របស់វាមិនមែនជាអត្ថប្រយោជន៍បន្ថែមទេ លុះត្រាតែមានភស្តុតាងឯករាជ្យ។
- **ការលុបចោលថ្លៃចំណាយដែលត្រូវបានផ្លាស់ទី។** គម្រោងដែលផ្លាស់ការកកស្ទះពីផ្លូវមួយទៅផ្លូវមួយទៀត ឬផ្លាស់ការក្លែងបន្លំពីឆានែលមួយទៅឆានែលមួយទៀត មិនទាន់បានបង្កើតអត្ថប្រយោជន៍សុទ្ធដូចដែល NPSV ចម្បងបង្ហាញទេ សូមមើល [ការបំលាស់ទី និងការសន្មតផល](../ការបំលាស់ទី-និងការសន្មតផល/)។
- **ការប្រើអត្រាបញ្ចុះតម្លៃឯកជន។** ការប្រើថ្លៃចំណាយដើមទុនពាណិជ្ជកម្ម (ឧទាហរណ៍ 8–10%) ជំនួសឱ្យអត្រាបញ្ចុះតម្លៃសង្គម ធ្វើឱ្យអត្ថប្រយោជន៍សាធារណៈរយៈពេលវែង ដូចជា ផលសុខភាព និងបរិស្ថាន ទាបជាងការពិតជាប្រព័ន្ធ សូមមើល [អត្រាបញ្ចុះតម្លៃសង្គម](../អត្រាបញ្ចុះតម្លៃសង្គម/)។
- **ការបម្លែងជាប្រាក់នូវអ្វីដែលមិនចម្រូងចម្រាស ហើយបំភ្លឺដោយដៃនូវអ្វីដែលចម្រូងចម្រាស។** ប្រសិនបើពីរភាគបីនៃអត្ថប្រយោជន៍របស់សំណើមួយ គឺជាការសន្សំប្រសិទ្ធភាពដែលបានបម្លែងជាប្រាក់ដោយទំនុកចិត្ត ហើយមួយភាគបីគឺជាផលសុខុមាលភាពដែលបានបម្លែងជាប្រាក់ដោយភាពរង្គោះរង្គើ នោះ NPSV ចម្បងលាយលេខរឹងជាមួយលេខទន់ដោយស្ងាត់ស្ងៀម ដូច្នេះត្រូវរាយការណ៍ដាច់ដោយឡែក។
