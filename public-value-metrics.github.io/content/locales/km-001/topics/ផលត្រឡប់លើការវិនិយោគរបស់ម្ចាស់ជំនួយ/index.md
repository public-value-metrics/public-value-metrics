# ផលត្រឡប់លើការវិនិយោគរបស់ម្ចាស់ជំនួយ

ផលត្រឡប់លើការវិនិយោគរបស់ម្ចាស់ជំនួយ (donor ROI) គឺជាអ្វីដែលផោនរបស់ម្ចាស់ជំនួយជាក់លាក់ម្នាក់ទិញបានពិតប្រាកដជាលទ្ធផល — មិនមែនសមាមាត្រប្រតិបត្តិការរបស់អង្គការសប្បុរសធម៌ ហើយក៏មិនមែនផលត្រឡប់របស់អង្គការសប្បុរសធម៌លើថវិកាសរុបរបស់ខ្លួនដែរ។ វាបកស្រាយ ROI ឡើងវិញពីទស្សនៈរបស់អង្គការ (យើងដំណើរការមានប្រសិទ្ធភាពប៉ុណ្ណា) ទៅទស្សនៈរបស់ម្ចាស់ជំនួយ (ការរួមចំណែកទីបំផុតរបស់ខ្ញុំផ្លាស់ប្តូរអ្វី) ហើយលេខទាំងពីរត្រូវបានចាត់ទុកជាប្រចាំ និងខុសថាជារឿងតែមួយ។

## ហេតុអ្វីបានជាសំខាន់

"ROI" របស់អង្គការសប្បុរសធម៌ផ្ទាល់ ក្នុងកម្រិតដែលឃ្លានេះត្រូវបានប្រើ ជាធម្មតាពណ៌នាអ្វីមួយដូចជា [តម្លៃក្នុងមួយអ្នកទទួលផល](../ថ្លៃចំណាយក្នុងមួយអ្នកទទួលផល/) ឬ [សមាមាត្រចំណាយរដ្ឋបាលរបស់អង្គការសប្បុរសធម៌](../សមាមាត្រចំណាយរដ្ឋបាលរបស់អង្គការសប្បុរសធម៌/) — រង្វាស់ប្រសិទ្ធភាពអង្គការ។ ROI របស់ម្ចាស់ជំនួយជាសំណួរខុសគ្នាទាំងស្រុង៖ ដោយសារអង្គការសប្បុរសធម៌នេះមានប្រាក់ចំណូលផ្សេងទៀតរួចហើយ តើប្រាក់របស់ម្ចាស់ជំនួយ*នេះ*បន្ថែមអ្វីនៅទីបំផុត? ប្រសិនបើអង្គការសប្បុរសធម៌នឹងផ្តល់កម្មវិធីដដែលទោះមាន ឬគ្មានអំណោយ £10,000 ជាក់លាក់ — ពីព្រោះវាមានទុនបម្រុងគ្រប់គ្រាន់ ឬព្រោះម្ចាស់មូលនិធិផ្សេងនឹងបំពេញគម្លាត — នោះ ROI របស់ម្ចាស់ជំនួយនៃអំណោយនោះជិតសូន្យ ទោះបីសមាមាត្រចំណាយរដ្ឋបាលរួម ឬតម្លៃក្នុងមួយលទ្ធផលរបស់អង្គការមើលទៅល្អប៉ុនណាក៏ដោយ។

នេះគឺជាសំណួរភាពបន្ថែម (additionality) ដូចគ្នាដែលជាមូលដ្ឋាននៃការវាយតម្លៃ [តម្លៃសមស្របនឹងប្រាក់ចំណាយ](../តម្លៃសមស្របនឹងថ្លៃចំណាយ/) ក្នុងការចំណាយសាធារណៈរបស់ចក្រភពអង់គ្លេស និង [ភាពបន្ថែម និង deadweight](../ភាពបន្ថែម-និងផលដែលនឹងកើតឡើងដោយខ្លួនឯង/) ក្នុងការវាយតម្លៃកម្មវិធី៖ តម្លៃដែលបានបង្កើតអាចទទួលក្រេឌីតដោយអ្នកផ្តល់មូលនិធិបានតែក្នុងកម្រិតដែលវាមិនកើតឡើងដោយខ្លួនឯង។ វេទិកាម្ចាស់ជំនួយធំៗ និងអង្គការផ្តល់ជំនួយប្រកបដោយប្រសិទ្ធភាព (Giving What We Can, GiveWell) សង់អនុសាសន៍របស់ខ្លួនដោយច្បាស់លាស់លើការបែងចែកនេះ ដោយសួរមិនមែនថា "តើនេះជាអង្គការសប្បុរសធម៌ល្អទេ" ទេ ប៉ុន្តែថា "តើអង្គការសប្បុរសធម៌នេះមានកន្លែងទំនេរសម្រាប់មូលនិធិបន្ថែម ដែលធ្វើឱ្យអំណោយរបស់ខ្ញុំជាផលបន្ថែមឬទេ"។

## គណិតវិទ្យា

```
Donor ROI ≠ Charity's operating efficiency

Donor ROI  ≈  (Outcome achieved with the gift) − (Outcome that would have
               occurred without it, i.e. the counterfactual)
             ─────────────────────────────────────────────────
                              Size of the gift

Key inputs:
  - Room for more funding (is the charity funding-constrained at the margin?)
  - Funging (would another donor have filled the gap?)
  - Marginal cost-effectiveness at the specific funding level (costs often
    rise as an intervention scales past its easiest-to-reach population)
```

សូមមើល [ប្រសិទ្ធភាពចំណាយនៃការប្រកាន់ខ្ជាប់ការសប្បុរសធម៌ប្រកបដោយប្រសិទ្ធភាព](../ប្រសិទ្ធភាពចំណាយនៃការសប្បុរសធម៌ប្រកបដោយប្រសិទ្ធភាព/) ស្តីពីរបៀបដែល GiveWell អនុវត្តសំណួរ "កន្លែងទំនេរសម្រាប់មូលនិធិបន្ថែម" និង [ការវិភាគស្ថានភាពប្រៀបធៀប](../ការវិភាគ-counterfactual/) សម្រាប់វិធីសាស្ត្រទូទៅ។

## ឧទាហរណ៍គណនា

ម្ចាស់ជំនួយម្នាក់កំពុងជ្រើសរើសរវាងអំណោយ £5,000 ពីរ៖

- **អង្គការសប្បុរសធម៌ C**៖ មានកម្មវិធីស្នូលដែលទទួលមូលនិធិពេញលេញ ជាមួយទុនបម្រុង £2 លាន និងបញ្ជីរង់ចាំនៃម្ចាស់មូលនិធិ។ £5,000 ទីបំផុតទំនងជាត្រូវបានបន្ថែមទៅទុនបម្រុង ឬសកម្មភាពអាទិភាពទាបជាង។ លទ្ធផលបន្ថែមដែលប៉ាន់ស្មានដោយម្ចាស់ជំនួយ៖ តិចតួចបំផុត — ប្រាក់នេះមិនប្តូរអ្វីដែលកើតឡើងយ៉ាងច្បាស់ទេ។
- **អង្គការសប្បុរសធម៌ D**៖ កម្មវិធីតូចដែលមានភស្តុតាងគាំទ្រ ដែលបានប្រកាសជាសាធារណៈថានឹងត្រូវបដិសេធមនុស្ស 200 នាក់ក្នុងត្រីមាសក្រោយ បើគ្មាន £50,000 បន្ថែម ហើយបានប្រមូល £42,000 នៃចំនួននោះរួចហើយ។ £5,000 ទីបំផុតទំនងខ្លាំងនឹងផ្តល់មូលនិធិដល់ការចែកចាយបន្ថែមពិតប្រាកដ — ឧទាហរណ៍ បម្រើមនុស្សបន្ថែម 20 នាក់ តាមតម្លៃក្នុងមួយអ្នកទទួលផលដែលអង្គការប្រកាសដោយខ្លួនឯងគឺ £250។

ទំហំអំណោយដូចគ្នា ម្ចាស់ជំនួយដូចគ្នា ប៉ុន្តែ ROI របស់ម្ចាស់ជំនួយខុសគ្នាខ្លាំង — មិនមែនពីព្រោះអង្គការសប្បុរសធម៌ C ជាអង្គការអន់ជាងទេ (វាប្រហែលជាមានតួលេខតម្លៃក្នុងមួយលទ្ធផលល្អជាងជាទូទៅ) ប៉ុន្តែពីព្រោះគម្លាតមូលនិធិទីបំផុតរបស់វាត្រូវបានបិទរួចហើយ។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

វេទិកាម្ចាស់ជំនួយ និងឧបករណ៍ផ្តល់អនុសាសន៍ការផ្តល់ជំនួយ ច្រើនពេកបង្ហាញតែរង្វាស់ប្រសិទ្ធភាពកម្រិតអង្គការ (សមាមាត្រចំណាយរដ្ឋបាល តម្លៃក្នុងមួយអ្នកទទួលផល) ពីព្រោះនោះជាអ្វីដែលអង្គការសប្បុរសធម៌បោះពុម្ពផ្សាយក្នុងរបាយការណ៍ប្រចាំឆ្នាំ និងជាអ្វីដែលងាយបំផុតក្នុងការទាញចូលតារាងប្រៀបធៀប។ ការបង្ហាញ ROI របស់ម្ចាស់ជំនួយឱ្យបានត្រឹមត្រូវតម្រូវឱ្យមានចំណុចទិន្នន័យខុសគ្នា និងពិបាករកប្រភពជាង៖ គម្លាតមូលនិធិបច្ចុប្បន្នដែលអង្គការសប្បុរសធម៌ប្រកាស ឬ "កន្លែងទំនេរសម្រាប់មូលនិធិបន្ថែម" ដែលផ្លាស់ប្តូរពេញមួយឆ្នាំ ហើយកម្រជាទិន្នន័យមានរចនាសម្ព័ន្ធ។ វេទិកាដែលចង់គាំទ្រហេតុផល ROI របស់ម្ចាស់ជំនួយពិតប្រាកដ ត្រូវការ ឬខ្សែផ្តល់ទិន្នន័យផ្ទាល់ពីការបង្ហាញគម្លាតមូលនិធិ (ដូចដែល GiveWell រក្សាដោយដៃសម្រាប់អង្គការសប្បុរសធម៌ដែលបានណែនាំ) ឬការបដិសេធច្បាស់លាស់ថាតារាងប្រៀបធៀបកំពុងបង្ហាញប្រសិទ្ធភាពអង្គការ មិនមែនភាពបន្ថែមរបស់ម្ចាស់ជំនួយទេ។ សូមមើល [សមាមាត្រចំណាយរដ្ឋបាលរបស់អង្គការសប្បុរសធម៌](../សមាមាត្រចំណាយរដ្ឋបាលរបស់អង្គការសប្បុរសធម៌/) សម្រាប់រង្វាស់ដែល ROI របស់ម្ចាស់ជំនួយច្រើនតែត្រូវបានច្រឡំជាមួយ។

## កំហុសដែលតែងកើតមាន

- **ច្រឡំប្រសិទ្ធភាពអង្គការសប្បុរសធម៌ជាមួយភាពបន្ថែមរបស់ម្ចាស់ជំនួយ។** អង្គការសប្បុរសធម៌ដែលដំណើរការល្អ និងមានចំណាយរដ្ឋបាលទាប នៅតែអាចមាន ROI ទីបំផុតរបស់ម្ចាស់ជំនួយជិតសូន្យ ប្រសិនបើវាមិនមានកង្វះមូលនិធិ។
- **មិនអើពើ funging។** ប្រសិនបើអ្នកផ្តល់មូលនិធិស្ថាប័នធំនឹងគ្របដណ្តប់គម្លាតទោះយ៉ាងណា អំណោយរបស់ម្ចាស់ជំនួយបុគ្គលម្នាក់ជំនួសប្រាក់របស់អ្នកផ្តល់មូលនិធិនោះ ជាជាងបន្ថែមការចែកចាយថ្មី។
- **សន្មតប្រសិទ្ធភាពចំណាយជាលីនេអ៊ែរនៅមាត្រដ្ឋានធំ។** អ្នកទទួលផលដែលទៅដល់ថោកបំផុតច្រើនតែត្រូវបានបម្រើមុន។ តម្លៃទីបំផុតក្នុងមួយលទ្ធផលច្រើនតែកើនឡើងនៅពេលកម្មវិធីពង្រីក ដូច្នេះ ROI លើផោនបន្ទាប់មិនដូចគ្នានឹង ROI លើផោនមធ្យមដែលបានចំណាយរួចហើយទេ។
- **គ្មានគម្លាតមូលនិធិដែលបានប្រកាស។** អង្គការសប្បុរសធម៌ ឬវេទិកាដែលមិនអាចនិយាយថា £X បន្ទាប់នឹងផ្តល់មូលនិធិអ្វី មិនអាចគាំទ្រការអះអាង ROI របស់ម្ចាស់ជំនួយពិតប្រាកដបានទេ អាចគាំទ្របានតែការអះអាងតម្លៃមធ្យមប៉ុណ្ណោះ។
