# ថ្លៃចំណាយឱកាសក្នុងការចំណាយសាធារណៈ

ថ្លៃចំណាយឱកាស (opportunity cost) គឺជាតម្លៃនៃជម្រើសជំនួសដ៏ល្អបំផុតដែលត្រូវបោះបង់ចោល នៅពេលស្ថាប័នសាធារណៈប្តូរផ្តាច់ប្រាក់ ពេលវេលាបុគ្គលិក
ឬដើមទុននយោបាយទៅលើជម្រើសមួយជំនួសឱ្យជម្រើសមួយទៀត។ នៅក្នុងនាយកដ្ឋានដែលមានថវិកាកំណត់ រាល់ប្រាក់មួយផោនដែលចំណាយលើកម្មវិធីមួយ
គឺជាប្រាក់មួយផោនដែលមិនអាចចំណាយលើកម្មវិធីល្អបំផុតបន្ទាប់បាន — ថ្លៃពិតនៃការសម្រេចចិត្តមិនមែនជាអ្វីដែលវាចំណាយទេ ប៉ុន្តែជាអ្វីដែលវាផ្លាស់ទីចេញ។

## ហេតុអ្វីបានជាសំខាន់

ថវិកាសាធារណៈមានដែនកំណត់សាច់ប្រាក់ក្នុងរយៈពេលនៃការពិនិត្យការចំណាយ ដូច្នេះ — មិនដូចក្រុមហ៊ុនឯកជនដែលកំពុងរីកចម្រើនទេ — នាយកដ្ឋានរដ្ឋាភិបាលមិនអាច "រកប្រាក់បន្ថែម" សម្រាប់គំនិតល្អមួយបានដោយសាមញ្ញទេ។
ការផ្តល់មូលនិធិឱ្យវា មានន័យថាត្រូវបញ្ឈប់មូលនិធិសម្រាប់អ្វីផ្សេង។ សៀវភៅ Green Book របស់ HM Treasury ចាត់ទុកនេះជាមូលដ្ឋានគ្រឹះ៖ រាល់ការវាយតម្លៃទាមទារឱ្យប្រៀបធៀបអន្តរាគមន៍មួយជាមួយមូលដ្ឋាន "do minimum" (ធ្វើតិចបំផុត)
*និង*ជាមួយការប្រើប្រាស់ជំនួសជាក់ស្តែងនៃធនធានដូចគ្នា ពិតជាព្រោះសំណួរពិតដែលក្រុមចំណាយរបស់ Treasury សួរ មិនដែលជា "តើនេះល្អទេ?" ទេ ប៉ុន្តែ "តើនេះល្អជាងអ្វីដែលប្រាក់នេះអាចទិញបានផ្សេងទៀតឬទេ?"។
គោលការណ៍វាយតម្លៃស្នូលរបស់ Green Book — ថាធនធានសាធារណៈគួរហូរទៅរកអន្តរាគមន៍ដែលមានតម្លៃសង្គមសុទ្ធខ្ពស់បំផុតក្នុងមួយផោន — គឺជាថ្លៃចំណាយឱកាសដែលបានចែងជាគោលនយោបាយ។

រឿងនេះងាយនិយាយ ប៉ុន្តែពិបាកអនុវត្ត ព្រោះ "ជម្រើសជំនួសល្អបំផុតបន្ទាប់" កម្រមើលឃើញក្នុងករណីអាជីវកម្មតែមួយ។ កម្មវិធីជំនួយ £2 លានសម្រាប់ការងារយុវជន
ត្រូវបានប្រៀបធៀបក្នុងករណីអាជីវកម្មជាមួយការមិនធ្វើអ្វីសោះ — ប៉ុន្តែគូប្រៀបធៀបស្មោះត្រង់គឺអន្តរាគមន៍ការងារយុវជនល្អបំផុតបន្ទាប់
ឬពិតប្រាកដជាងនេះ ការប្រើប្រាស់ £2 លានល្អបំផុតបន្ទាប់នៅកន្លែងណាមួយក្នុងផតហ្វូលីយ៉ូ រួមទាំងការចំណាយក្រៅពីការងារ។
Magenta Book (HM Treasury, 2020) ព្រមានយ៉ាងច្បាស់ថា ការវាយតម្លៃដែលប្រៀបធៀប "ជាមួយអន្តរាគមន៍" ទៅនឹង "គ្មានអន្តរាគមន៍" ប៉ាន់ស្មានទាបពេកនូវកម្រិតដែលអន្តរាគមន៍ត្រូវឆ្លងកាត់
ព្រោះ "ដោយគ្មានអន្តរាគមន៍នេះ" មិនដូចគ្នានឹង "ជាមួយគ្មានអ្វីសោះ" ទេ — ប្រាក់ដែលត្រូវបានដោះលែងផ្តល់មូលនិធិដល់អ្វីផ្សេង។

## គណិតវិទ្យា

```
ថ្លៃចំណាយឱកាសនៃការជ្រើស A = តម្លៃនៃជម្រើសជំនួសល្អបំផុតដែលបោះបង់ចោល B

តម្លៃសាធារណៈសុទ្ធនៃ A = តម្លៃ(A) − តម្លៃ(B) មិនមែន តម្លៃ(A) − 0 ទេ
```

គ្មានរូបមន្តសកលទេ ព្រោះជម្រើសជំនួសដែលបោះបង់ចោលអាស្រ័យលើបរិបទ ប៉ុន្តែវិន័យនេះអាចយកទៅប្រើទូទៅបាន៖ កំណត់ការប្រើប្រាស់ល្អបំផុតបន្ទាប់ជាក់ស្តែងនៃបន្ទាត់ថវិកាដូចគ្នា (មិនមែន "មិនធ្វើអ្វី" ដ៏ឧត្តមគតិទេ) វាយតម្លៃវាលើមូលដ្ឋានដូចគ្នា (បម្លែងជាប្រាក់ ប្រសិនបើអាច តាម
[ការវិភាគថ្លៃដើម-ផលប្រយោជន៍សង្គម](../ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម/)) ហើយដក។

## ឧទាហរណ៍គណនា

**បន្ទាត់ថវិកានាយកដ្ឋាន**៖ មូលនិធិបំលែងឌីជីថល £5 លាន អាចផ្តល់មូលនិធិបានតែមួយក្នុងចំណោមសំណើពីរប៉ុណ្ណោះក្នុងឆ្នាំហិរញ្ញវត្ថុនេះ។

- *ជម្រើស A*៖ វេទិកាគ្រប់គ្រងករណីថ្មី អត្ថប្រយោជន៍ដែលបម្លែងជាប្រាក់ £7.2 លាន ក្នុងរយៈពេល 5 ឆ្នាំ
  (សន្សំប្រសិទ្ធភាព បូកនឹងការដោះស្រាយករណីលឿនជាងមុន)។
- *ជម្រើស B*៖ សេវាផ្ទៀងផ្ទាត់អត្តសញ្ញាណដែលចែករំលែករវាងនាយកដ្ឋានបី អត្ថប្រយោជន៍ដែលបម្លែងជាប្រាក់ £6.4 លាន ក្នុងរយៈពេល 5 ឆ្នាំ។

ករណីអាជីវកម្មឆោតល្ងង់សម្រាប់ A ប្រៀបធៀបអត្ថប្រយោជន៍ £7.2 លាន ទៅនឹងថ្លៃដើម £5 លាន ហើយរាយការណ៍សមាមាត្រអត្ថប្រយោជន៍ទៅថ្លៃដើម 1.44:1 — ហាក់ដូចជាខ្លាំង។ ប៉ុន្តែដោយសារ A និង B ប្រកួតប្រជែងគ្នាសម្រាប់ប្រាក់ £5 លានដូចគ្នា
ថ្លៃចំណាយឱកាសនៃការជ្រើស A គឺអត្ថប្រយោជន៍ £6.4 លានរបស់ B ដែលបោះបង់ចោល។ ករណី*សុទ្ធ*សម្រាប់ A លើជម្រើសជំនួសជាក់ស្តែងគឺត្រឹមតែ £7.2m − £6.4m = £0.8 លាន មិនមែនតួលេខចំណងជើងពេញ £7.2 លានទេ។ ប្រសិនបើជម្រើសទីបី C
ផ្តល់អត្ថប្រយោជន៍ £7.5 លានសម្រាប់ប្រាក់ £5 លានដូចគ្នា ការផ្តល់មូលនិធិឱ្យ A ជំនួស C នឹងបំផ្លាញតម្លៃសាធារណៈ £0.3 លាន ទោះបីជាករណីអាជីវកម្មរបស់ A ផ្ទាល់មើលទៅមានភាពសមហេតុផលពេញលេញនៅពេលដាច់ដោយឡែកក៏ដោយ។

**ពេលវេលាបុគ្គលិករបស់អាជ្ញាធរមូលដ្ឋាន**៖ ក្រុមទិន្នន័យបីនាក់របស់ក្រុមប្រឹក្សាអាចសាងសង់ផ្ទាំងគ្រប់គ្រងបញ្ជីរង់ចាំលំនៅដ្ឋាន

ថ្លៃពិតនៃការសាងសង់ផ្ទៃក្នុង "ឥតគិតថ្លៃ" គឺអត្ថប្រយោជន៍ធំជាងឆ្ងាយដែលក្រុមអាចបង្កើតបាននៅកន្លែងផ្សេង។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

សមត្ថភាពវិស្វកម្មនៅក្នុងស្ថាប័នសាធារណៈគឺជាថវិកាដែលមានដែនកំណត់ខ្លួនឯង — សមត្ថភាពស្ព្រីន មិនមែនផោនទេ — ហើយវិន័យដូចគ្នាអនុវត្តដោយផ្ទាល់៖

- តែងតែដាក់ឈ្មោះគូប្រៀបធៀប៖ ករណីអាជីវកម្មរបស់លក្ខណៈពិសេសមួយគួរចែងថាតើអ្វីផ្សេងទៀតដែលសប្តាហ៍ក្រុមដូចគ្នាអាចប្រគល់ជូនបាន មិនមែនត្រឹមតែអត្ថប្រយោជន៍របស់ខ្លួនទេ។
- ចាត់ទុក "យើងមានសមត្ថភាពវិស្វកម្មសល់" ជាការចាប់ផ្តើមនៃការវិភាគថ្លៃចំណាយឱកាស មិនមែនទីបញ្ចប់ទេ — សមត្ថភាពសល់នៅតែមានការប្រើប្រាស់ជំនួសល្អបំផុត ទោះបីការប្រើនោះជាការសងបំណុលបច្ចេកទេសក៏ដោយ
  (សូមមើល [បំណុលបច្ចេកទេសជាការហិនហោចតម្លៃសាធារណៈ](../បំណុលបច្ចេកទេសជាការកាត់បន្ថយតម្លៃសាធារណៈ/))។
- ភ្ជាប់រឿងនេះដោយផ្ទាល់ទៅ[តម្លៃសមស្របនឹងថវិកា](../តម្លៃសមស្របនឹងថ្លៃចំណាយ/)៖ តេស្ត "សេដ្ឋកិច្ច" របស់ VFM គ្មានន័យអ្វីទេ បើគ្មានគូប្រៀបធៀបថ្លៃចំណាយឱកាសដ៏ស្មោះត្រង់ និងទៅ
  [ថ្លៃដើមនៃការពន្យារពេលក្នុងកម្មវិធីសាធារណៈ](../ថ្លៃចំណាយនៃការយឺតយ៉ាវក្នុងកម្មវិធីសាធារណៈ/) ដែលកំណត់តម្លៃវិមាត្រពេលវេលានៃតក្កវិជ្ជាជម្រើសជំនួសដែលបោះបង់ចោលដូចគ្នា។

## កំហុសដែលតែងកើតមាន

- **ប្រៀបធៀបជាមួយ "មិនធ្វើអ្វី" ជំនួសឱ្យជម្រើសជំនួសល្អបំផុតបន្ទាប់។** Green Book ទាមទារមូលដ្ឋាន "do minimum" ពិតជាព្រោះថ្លៃចំណាយឱកាសពិតកម្រស្មើសូន្យ។
  ករណីអាជីវកម្មដែលឆ្លងតែកម្រិត "មិនធ្វើអ្វី" មិនទាន់បានបង្ហាញថាវាយកឈ្នះជម្រើសជំនួសជាក់ស្តែងទេ។
- **មិនអើពើនឹងការប្រកួតប្រជែងឆ្លងនាយកដ្ឋានសម្រាប់ថវិកាដូចគ្នា។** បន្ទាត់ថវិកាដែលមើលទៅដូចបានបម្រុងទុកក្នុងនាយកដ្ឋានមួយ
  ជារឿយៗប្រកួតប្រជែងនៅកម្រិតខ្ពស់ជាង (ការពិនិត្យការចំណាយ កម្មវិធីដើមទុន) ជាកន្លែងដែលថ្លៃចំណាយឱកាសពិតត្រូវបានសម្រេច។
- **សន្មតថាពេលវេលាបុគ្គលិកដែលដោះលែងមិនមានតម្លៃបន្ថែមទៀត។** ពេលវេលាដែល "សន្សំបាន" បង្កើតតម្លៃតែនៅពេលដែលត្រូវបានផ្លាស់ប្តូរទៅអ្វីដែលមានតម្លៃ
  ប្រសិនបើគ្មានការប្រើជំនួសនោះទេ ការសន្សំគឺគ្រាន់តែជាទ្រឹស្តី។
