# ការវាយតម្លៃតាមចំណូលចិត្តដែលបានបង្ហាញ

វិធីសាស្ត្រចំណូលចិត្តដែលបានបង្ហាញ (revealed preference) សន្និដ្ឋានតម្លៃនៃទំនិញក្រៅទីផ្សារពីឥរិយាបថដែលអាចសង្កេតបានក្នុងទីផ្សារដែលទាក់ទង ជាជាងសួរមនុស្សដោយផ្ទាល់។ ការកំណត់តម្លៃតាមលក្ខណៈ (hedonic pricing) និងវិធីសាស្ត្រថ្លៃធ្វើដំណើរ (travel-cost method) គឺជាបច្ចេកទេសធ្វើការសំខាន់ពីរ៖ ទាំងពីរចាប់ផ្តើមពីប្រតិបត្តិការពិត ហើយគណនាថយក្រោយនូវតម្លៃបង្កប់សម្រាប់អ្វីដែលមិនដែលត្រូវបានលក់ដោយផ្ទាល់។

## ហេតុអ្វីបានជាសំខាន់

ខណៈដែលវិធីសាស្ត្រ[ចំណូលចិត្តដែលបានប្រកាស](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស/) សួរសំណួរសម្មតិកម្ម វិធីសាស្ត្រចំណូលចិត្តដែលបានបង្ហាញ សង្កេតអ្វីដែលមនុស្សបានបង់ជាក់ស្តែង ដែល Green Book ចាត់ទុកថាជាភស្តុតាងដែលអាចជឿជាក់បានជាងជាទូទៅ ប្រសិនបើអ្វីៗផ្សេងទៀតស្មើគ្នា ព្រោះវាមិនទទួលរងភាពលំអៀងសម្មតិកម្ម —

វិធីសាស្ត្រថ្លៃធ្វើដំណើរធ្វើរឿងស្រដៀងគ្នាសម្រាប់ទីតាំងកម្សាន្តដែលគ្មានថ្លៃចូល៖ ពេលវេលា និងប្រាក់ដែលមនុស្សចំណាយក្នុងការធ្វើដំណើរទៅទីតាំងបង្ហាញបន្ទាត់ខាងក្រោមនៃតម្លៃរបស់ទីតាំងសម្រាប់ពួកគេ ព្រោះគ្មាននរណាចំណាយថ្លៃលើសពីតម្លៃដែលការទស្សនាមានសម្រាប់ខ្លួន។

វិធីសាស្ត្រទាំងពីរមានដែនកំណត់រចនាសម្ព័ន្ធដូចគ្នា៖ ពួកវាអាចវាយតម្លៃបានតែអ្វីដែលបង្កប់ក្នុងប្រតិបត្តិការទីផ្សារដែលមានស្រាប់ប៉ុណ្ណោះ។ សំឡេងរំខានជិតផ្លូវរត់យន្តហោះបង្ហាញក្នុងតម្លៃផ្ទះ ព្រោះមនុស្សដែលខ្វល់ពីសំឡេងរំខានជ្រើសរើសលំនៅដ្ឋានស្ងាត់ជាង។ តម្លៃអត្ថិភាពនៃប្រភេទសត្វដែលគ្មាននរណាទស្សនា ឬរស់នៅជិត មិនបង្ហាញក្នុងប្រតិបត្តិការណាមួយទាល់តែសោះ ដែលជាគម្លាតពិតប្រាកដដែលវិធីសាស្ត្រ[ចំណូលចិត្តដែលបានប្រកាស](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស/) មានដើម្បីបំពេញ។

  តម្លៃផ្ទះ = f(លក្ខណៈរចនាសម្ព័ន្ធ លក្ខណៈទីតាំង

## ឧទាហរណ៍គណនា

**រដ្ឋាភិបាលជាតិ**៖ វិធីសាស្ត្រតម្លៃស្រមោលកាបូនរបស់ Green Book ផ្ទាល់ពឹងផ្អែកមួយផ្នែកលើភស្តុតាង hedonic ប៉ុន្តែករណីឧទាហរណ៍សាមញ្ញជាងគឺសំឡេងរំខានយន្តហោះ។ ការសិក្សា hedonic ដែលតំរែតំរង់តម្លៃលក់ផ្ទះក្នុងតំបន់ផ្លូវហោះហើរទៅនឹងការប៉ះពាល់សំឡេងរំខានដែលមានទម្ងន់តាមចម្ងាយ ដោយគ្រប់គ្រងទំហំ អាយុ និងតំបន់ទទួលសាលា រកឃើញថាការកើនឡើងមធ្យមនៃការប៉ះពាល់សំឡេង 1 ដេស៊ីបែលនីមួយៗទាក់ទងនឹងការថយចុះតម្លៃផ្ទះ 0.5%។ សម្រាប់ផ្ទះធម្មតា £280,000 ក្នុងតំបន់ដែលរងផលប៉ះពាល់៖

```
តម្លៃបង្កប់ក្នុងមួយដេស៊ីបែល = £280,000 × 0.5% = £1,400 ក្នុងមួយគ្រួសារ
គ្រួសាររងផលប៉ះពាល់ពីការកើនឡើង 3dB ពីផ្លូវរត់យន្តហោះថ្មី = 18,000
ថ្លៃដើមបង្កប់សរុបនៃការកើនឡើងសំឡេងរំខាន = £1,400 × 3 × 18,000 = £75.6m
```

នេះគឺជាថ្លៃដើមធ្វើជាដើមទុនម្តងតែមួយ (បង្កប់ក្នុងតម្លៃផ្ទះ) ដែលការវាយតម្លៃត្រូវប្រុងប្រយ័ត្នកុំរាប់ទ្វេដងប្រឆាំងនឹងចរន្តថ្លៃដើមរំខានសំឡេងប្រចាំឆ្នាំដែលបានប៉ាន់ស្មានដាច់ដោយឡែក។

**សប្បុរសធម៌**៖ សប្បុរសធម៌បរិស្ថានមួយប្រើវិធីសាស្ត្រថ្លៃធ្វើដំណើរដើម្បីវាយតម្លៃតំបន់ការពារធម្មជាតិដែលចូលដោយឥតគិតថ្លៃ។ ទិន្នន័យស្ទង់មតិលើលេខកូដប្រៃសណីយ៍អ្នកទស្សនា ផ្តល់ថ្លៃធ្វើដំណើរទៅមកមធ្យម (ពេលវេលាត្រូវបានវាយតម្លៃតាមតម្លៃពេលវេលាក្រៅការងារដែល Green Book ណែនាំ បូកឥន្ធនៈ) £14 ក្នុងមួយការទស្សនា ជាមួយការទស្សនា 40,000 ក្នុងមួយឆ្នាំ។ ខ្សែកោងតម្រូវការដែលបានប៉ាន់ស្មាន — អត្រាទស្សនាធ្លាក់ចុះនៅពេលថ្លៃធ្វើដំណើរពីតំបន់មួយកើនឡើង — បង្កប់អតិរេកអ្នកប្រើប្រាស់ក្នុងមួយការទស្សនា លើសពី £14 ដែលបានចំណាយជាក់ស្តែង ប្រហែល £9។

```
តម្លៃប្រចាំឆ្នាំសរុប = 40,000 ការទស្សនា × (£14 ដែលបានចំណាយ + £9 អតិរេកអ្នកប្រើប្រាស់)
                    = 40,000 × £23 ≈ £920,000/ឆ្នាំ
```

នេះធំជាងចំណូលថ្លៃចូលសូន្យរបស់តំបន់ការពារ ហើយផ្តល់ឱ្យអ្នកគ្រប់គ្រងសប្បុរសធម៌នូវតួលេខដែលអាចការពារបានសម្រាប់តម្លៃកម្សាន្តនៃទីតាំង នៅពេលបង្ហាញករណីដល់ម្ចាស់មូលនិធិ។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

ការគិតតាមចំណូលចិត្តដែលបានបង្ហាញលេចឡើងក្នុងការវិភាគផលិតផលវិស័យសាធារណៈ ញឹកញាប់ជាងដែលអ្នកអនុវត្តដឹង៖ ទិន្នន័យការប្រើប្រាស់ពីសេវាឌីជីថលរដ្ឋាភិបាលឥតគិតថ្លៃ ខ្លួនវាជាភស្តុតាងចំណូលចិត្តដែលបានបង្ហាញនៃតម្លៃ (ភាពញឹកញាប់ រយៈពេលវគ្គ និង — គួរឱ្យចាប់អារម្មណ៍បំផុត — លំនាំប្រើប្រាស់ម្តងហើយម្តងទៀត ធៀបនឹងម្តងតែមួយ អាចវិភាគតាមរបៀបដូចគ្នាដែលម៉ូដែលថ្លៃធ្វើដំណើរចាត់ទុកភាពញឹកញាប់ទស្សនាធៀបនឹងចម្ងាយ)។ នៅពេលដែលសេវាមានជម្រើសជំនួសពិត (បណ្តាញក្រដាស ខ្សែទូរស័ព្ទ) "ថ្លៃ" ដែលប្រជាពលរដ្ឋជួបប្រទះដើម្បីប្រើបណ្តាញឌីជីថលជំនួស (ពេលវេលា ទិន្នន័យ ឧបករណ៍) អាចត្រូវបានប៉ាន់ស្មាន និងប្រៀបធៀបនឹងការប្រើប្រាស់ ដោយឆ្លុះបញ្ចាំងតក្កវិជ្ជាថ្លៃធ្វើដំណើរដោយផ្ទាល់។ សូមមើល
[ស្តង់ដារសេវាឌីជីថល](../ស្តង់ដារសេវាឌីជីថល/) និង
[តម្លៃនៃទិន្នន័យបើកចំហ](../តម្លៃនៃទិន្នន័យបើកចំហ/) ដែលប្រឈមនឹងបញ្ហាវាយតម្លៃដូចគ្នាសម្រាប់ទំនិញដែលគ្មានតម្លៃទីផ្សារផ្ទាល់។

## កំហុសដែលតែងកើតមាន

- **ភាពលំអៀងអថេរដែលបាត់ក្នុងម៉ូដែល hedonic។** ការមិនរាប់បញ្ចូលលក្ខណៈទាក់ទង (គុណភាពសាលាដែលទាក់ទងទាំងតម្លៃផ្ទះ និងអថេរបរិស្ថានដែលចាប់អារម្មណ៍) បង្កើតភាពលំអៀងលើការប៉ាន់ស្មានតម្លៃបង្កប់។ ការបញ្ជាក់ត្រូវតែត្រូវបានរាយការណ៍ និងពិនិត្យ មិនមែនត្រឹមតែលទ្ធផលទេ។
- **មិនអើពើនឹងទីតាំងជំនួសក្នុងការសិក្សាថ្លៃធ្វើដំណើរ។** តម្លៃដែលបានបង្ហាញរបស់អ្នកទស្សនាសម្រាប់ទីតាំងមួយ ត្រូវបានប៉ាន់ស្មានទាប ប្រសិនបើមានទីតាំងជំនួសជិតជាងមាន ហើយមិនត្រូវបានគ្រប់គ្រង — ពួកគេអាចទស្សនាភាគច្រើនព្រោះវាឥតគិតថ្លៃ មិនមែនព្រោះវាមានតម្លៃពិសេសទេ។
- **អនុវត្តចំណូលចិត្តដែលបានបង្ហាញលើទំនិញដែលគ្មានអេកូទីផ្សារទាល់តែសោះ។** តម្លៃអត្ថិភាព តម្លៃជម្រើស និងតម្លៃកេរ្តិ៍មរតក មិនបង្ហាញក្នុងប្រតិបត្តិការណាមួយទេ ហើយមិនអាចទាញយកបានដោយវិធីសាស្ត្រ hedonic ឬថ្លៃធ្វើដំណើរទេ — គម្លាតនោះជារបស់[ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស/)។
- **ច្រឡំតម្លៃដើមទុន (ម្តងតែមួយ) ជាមួយចរន្តប្រចាំឆ្នាំ។** ផលប៉ះពាល់តម្លៃផ្ទះ hedonic ជាធម្មតាជាតម្លៃដើមទុនម្តងតែមួយ។ ការចាត់ទុកវាជាចរន្តអត្ថប្រយោជន៍ប្រចាំឆ្នាំ បំផ្លើសការវាយតម្លៃ។
