# ការវាយតម្លៃតាមចំណូលចិត្តដែលបានប្រកាស

វិធីសាស្ត្រចំណូលចិត្តដែលបានប្រកាស (stated preference) ប៉ាន់ស្មានតម្លៃនៃទំនិញគ្មានទីផ្សារ ដោយសួរមនុស្សដោយផ្ទាល់ថា តើពួកគេនឹងមានឆន្ទៈបង់ប៉ុន្មានសម្រាប់វា ឬមានឆន្ទៈទទួលសំណងប៉ុន្មានដើម្បីលះបង់វា ជាទូទៅតាមរយៈការស្ទង់មតិមានរចនាសម្ព័ន្ធដែលពណ៌នាសេណារីយ៉ូសម្មតិកម្ម។ ការវាយតម្លៃតាមលក្ខខណ្ឌ (contingent valuation) គឺជាបច្ចេកទេសដែលល្បីបំផុតក្នុងគ្រួសារនេះ។

## ហេតុអ្វីបានជាសំខាន់

ឧបសម្ព័ន្ធ 2 របស់ Green Book (ការណែនាំបន្ថែមស្តីពីការវាយតម្លៃផលប៉ះពាល់គ្មានទីផ្សារ) គាំទ្រវិធីសាស្ត្រចំណូលចិត្តដែលបានប្រកាស សម្រាប់ទំនិញដែលគ្មានប្រតិបត្តិការទីផ្សារដែលអាចសង្កេតបានទាល់តែសោះ ដើម្បីសន្និដ្ឋានតម្លៃ ដូចជា គុណភាពខ្យល់ ជីវចម្រុះ ការការពារទឹកជំនន់ និងតម្លៃអត្ថិភាពនៃទេសភាពដែលនរណាម្នាក់ប្រហែលជាមិនដែលទៅទស្សនា
(<https://www.gov.uk/government/publications/the-green-book-appraisal-and-evaluation-in-central-government>)។
Defra បានចេញផ្សាយការណែនាំចំណូលចិត្តដែលបានប្រកាសផ្ទាល់ខ្លួនសម្រាប់ការវាយតម្លៃបរិស្ថាន ជាពិសេសពីព្រោះតម្លៃបរិស្ថានភាគច្រើន (ការអភិរក្សជម្រកធម្មជាតិ គុណភាពទឹក) គ្មានទីផ្សារតំណាងទាល់តែសោះ មិនដូចសំឡេងរំខាន ជាឧទាហរណ៍ ដែលយ៉ាងហោចណាស់ទាក់ទងនឹងតម្លៃផ្ទះដែលអាចសង្កេតបាន (សូមមើល
[ការវាយតម្លៃតាមចំណូលចិត្តដែលបានបង្ហាញ](../ការវាយតម្លៃតាមចំណូលចិត្តដែលបានបង្ហាញ/))។

ភាពទាក់ទាញស្នូលនៃចំណូលចិត្តដែលបានប្រកាស គឺវាអាចវាយតម្លៃអ្វីៗបានទាំងអស់ រួមទាំងទំនិញដែលគ្មាននរណាធ្លាប់ធ្វើប្រតិបត្តិការ ក៏ជាប្រភពនៃបញ្ហាភាពជឿជាក់របស់វាដែរ។ ដោយសារអ្នកឆ្លើយមិនបានចំណាយប្រាក់ពិតប្រាកដ ការស្ទង់មតិវាយតម្លៃតាមលក្ខខណ្ឌងាយរងនូវភាពលំអៀងសម្មតិកម្ម (មនុស្សប៉ាន់ឆន្ទៈក្នុងការបង់ហួសការពិត នៅពេលគ្មានដែនកំណត់ថវិកាពិតប្រាកដ) ឥទ្ធិពលបង្កប់ (ទំនិញដូចគ្នាត្រូវបានវាយតម្លៃខុសគ្នា អាស្រ័យលើអ្វីផ្សេងទៀតនៅក្នុងការស្ទង់មតិ) និងភាពលំអៀងចំណុចចាប់ផ្តើមក្នុងការរចនាល្បែងដេញថ្លៃ។ គណៈកម្មការ NOAA ឆ្នាំ 1993 ស្តីពីការវាយតម្លៃតាមលក្ខខណ្ឌ ដែលបានប្រជុំក្រោយជម្លោះតុលាការការហូរប្រេង Exxon Valdez បានកំណត់ស្តង់ដាររចនា គឺទម្រង់ប្រជាមតិ "តើអ្នកនឹងបង់ £X ឬទេ បាទ/ចាស ឬទេ" ជាជាងការដេញថ្លៃបើកចំហ និងការរំលឹកកាតព្វកិច្ចអំពីដែនកំណត់ថវិកាពិតរបស់អ្នកឆ្លើយ ដែលនៅតែជាស្តង់ដារយោងសម្រាប់ការស្ទង់មតិដែលអាចការពារបាន។

## គណិតវិទ្យា

```
ការវាយតម្លៃតាមលក្ខខណ្ឌ (ទម្រង់ប្រជាមតិ)៖
  បង្ហាញជម្រើសគោលពីរ៖ "តើអ្នកនឹងបង់ £X ក្នុងមួយឆ្នាំសម្រាប់លទ្ធផល Y ឬទេ? បាទ/ចាស ឬទេ"
  ប្តូរ X ដោយចៃដន្យក្នុងចំណោមអ្នកឆ្លើយ។
  ប៉ាន់ស្មានឆន្ទៈក្នុងការបង់ជាអនុគមន៍នៃអត្រាចម្លើយ បាទ/ចាស ឬទេ នៅ X នីមួយៗ។

WTP មធ្យម = ក្រឡាផ្ទៃក្រោមខ្សែកោងតម្រូវការដែលបានប៉ាន់ស្មាន
តម្លៃសរុប = WTP មធ្យម × ប្រជាជនដែលរងផលប៉ះពាល់

បំរែបំរួលការពិសោធជម្រើស (choice experiment, គំរូជម្រើសដាច់ដោយឡែក)៖
  បង្ហាញអ្នកឆ្លើយនូវជម្រើសម្តងហើយម្តងទៀតរវាងកញ្ចប់គុណលក្ខណៈ
  (រួមទាំងគុណលក្ខណៈថ្លៃចំណាយ) ប៉ាន់ស្មានតម្លៃបង្កប់សម្រាប់គុណលក្ខណៈ
  មិនមែនថ្លៃចំណាយនីមួយៗ ពីការជ្រើសរើសដែលអ្នកឆ្លើយបង្ហាញ។
```

បំរែបំរួលការពិសោធជម្រើសជាទូទៅត្រូវបានគេពេញចិត្តក្នុងការអនុវត្តបច្ចុប្បន្ននៃចក្រភពអង់គ្លេស ជាងការវាយតម្លៃតាមលក្ខខណ្ឌមួយសំណួរ ពីព្រោះការបង្ខំអ្នកឆ្លើយឱ្យជ្រើសរើសរវាងគុណលក្ខណៈជាច្រើនធៀបនឹងថ្លៃចំណាយម្តងហើយម្តងទៀត បង្កើតការប៉ាន់ស្មានដែលស៊ីសង្វាក់គ្នាផ្ទៃក្នុងជាង និងពិបាកបោកបញ្ឆោតជាងសំណួរ បាទ/ចាស ឬទេ តែមួយ។

## ឧទាហរណ៍គណនា

**រដ្ឋាភិបាលជាតិ**៖ Defra ចាត់ចែងការស្ទង់មតិវាយតម្លៃតាមលក្ខខណ្ឌ ដើម្បីវាយតម្លៃកម្មវិធីកែលម្អគុណភាពទឹកទន្លេ។ ការស្ទង់មតិទម្រង់ប្រជាមតិនៃគ្រួសារ 2,000 រកឃើញថា 62% នឹងបង់ £40/ឆ្នាំ តាមរយៈការបន្ថែមវិក្កយបត្រទឹកសម្មតិកម្ម ហើយខ្សែកោងតម្រូវការដែលបានប៉ាន់ស្មានផ្តល់ឆន្ទៈក្នុងការបង់មធ្យម £28/ឆ្នាំក្នុងមួយគ្រួសារ។

```
WTP មធ្យម = £28/គ្រួសារ/ឆ្នាំ
គ្រួសារក្នុងតំបន់ស្រោចស្រពទន្លេ = 340,000
តម្លៃប្រចាំឆ្នាំសរុប = £28 × 340,000 = £9.52m/ឆ្នាំ

ក្នុងរយៈពេលវាយតម្លៃ 20 ឆ្នាំ ក្នុងអត្រាបញ្ចុះតម្លៃ 3.5% (កត្តាប្រាក់បៀវត្សរ៍ ≈ 14.2)៖
PV(អត្ថប្រយោជន៍) ≈ £9.52m × 14.2 ≈ £135m
```

តួលេខសរុបនេះបន្ទាប់មកត្រូវបានប្រៀបធៀបនឹងផ្នែកថ្លៃចំណាយនៃ[ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម](../ការវិភាគថ្លៃដើម-អត្ថប្រយោជន៍សង្គម/)របស់កម្មវិធី។ Green Book តម្រូវឱ្យរាយការណ៍ភស្តុតាងចំណូលចិត្តដែលបានប្រកាសបែបនេះ រួមជាមួយចន្លោះទំនុកចិត្ត និងវិធីសាស្ត្រស្ទង់មតិរបស់វា មិនមែនជាការប៉ាន់ស្មានចំណុចទទេទេ ពីព្រោះលេខមូលដ្ឋានមានភាពផុយស្រួយជាងតម្លៃទីផ្សារ។

**អង្គការសប្បុរសធម៌**៖ ទុនបេតិកភណ្ឌមួយស្ទង់មតិអ្នកទស្សនា និងអ្នកមិនទស្សនាអំពីឆន្ទៈក្នុងការបង់ ដើម្បីការពារការបិទអគារប្រវត្តិសាស្ត្រដែលក្រុមទាំងពីរមិនចាំបាច់ទៅទស្សនា (តម្លៃអត្ថិភាពរបស់វា)។ ដោយសារអ្នកមិនទស្សនាដែលនឹងមិនដែលឃើញអគារនោះ នៅតែរាយការណ៍ WTP វិជ្ជមាន ការស្ទង់មតិចាប់យកតម្លៃអត្ថិភាព និងតម្លៃទុកជូនកូនចៅ ដែលការរាប់ចំណូលថ្លៃទស្សនាសាមញ្ញ (តំណាងចំណូលចិត្តដែលបានបង្ហាញ) នឹងខកខានទាំងស្រុង ដែលបង្ហាញពីអត្ថប្រយោជន៍ពិតប្រាកដនៃចំណូលចិត្តដែលបានប្រកាស នៅកន្លែងដែលគ្មានប្រតិបត្តិការទីផ្សារណាមួយ ដើម្បីបង្ហាញតម្លៃ។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

វិធីសាស្ត្រចំណូលចិត្តដែលបានប្រកាសកម្រអនុវត្តដោយផ្ទាល់ចំពោះការងារវិស្វកម្មកម្មវិធី ប៉ុន្តែវិស្វករដែលកសាងវេទិកាពិគ្រោះយោបល់ពលរដ្ឋ ឧបករណ៍ចូលរួមថវិកា ឬហេដ្ឋារចនាសម្ព័ន្ធស្ទង់មតិសាធារណៈ ជារឿយៗកំពុងកសាងឧបករណ៍ដែលសេដ្ឋកិច្ចសាស្ត្រពឹងផ្អែក។ ការធ្វើឱ្យព័ត៌មានលម្អិតនៃការរចនាការស្ទង់មតិត្រឹមត្រូវ គឺចំនួនទឹកប្រាក់ដេញថ្លៃចៃដន្យ ការរៀបចំប្រជាមតិគោលពីរជាជាងសំណួរបើកចំហ ការរំលឹកដែនកំណត់ថវិកាច្បាស់លាស់ មិនមែនជាភាពស្រស់ស្អាតនៃ UX ទេ វាជាអ្វីដែលធ្វើឱ្យការវាយតម្លៃលទ្ធផលអាចការពារបានក្រោមការត្រួតពិនិត្យ ការស្ទង់មតិក្នុងកម្មវិធីដែលរចនាមិនល្អអាចបំផ្លាញការវិភាគសេដ្ឋកិច្ចជាច្រើនខែបន្តបន្ទាប់។ សូមមើល [សូចនាករការពេញចិត្តរបស់ពលរដ្ឋ](../រង្វាស់ការពេញចិត្តរបស់ពលរដ្ឋ/)
សម្រាប់វិន័យទូទៅជាងអំពីការទាញយកទិន្នន័យមតិសាធារណៈដែលនឹងមានទម្ងន់វិភាគ។

## កំហុសដែលតែងកើតមាន

- **សំណួរបើកចំហ "តើអ្នកនឹងបង់ប៉ុន្មាន?"។** ទាំងនេះងាយរងភាពលំអៀងយុទ្ធសាស្ត្រ និងភាពលំអៀងយុថ្កាច្រើនជាងការរៀបចំប្រជាមតិគោលពីរ ហើយអនុសាសន៍របស់គណៈកម្មការ NOAA ឱ្យប្រើទម្រង់ប្រជាមតិ មានជាក់លាក់ពីព្រោះការទាញយកបើកចំហដំណើរការមិនល្អ។
- **គ្មានការរំលឹកអំពីដែនកំណត់ថវិកាពិតរបស់អ្នកឆ្លើយ។** បើគ្មានវា WTP ដែលបានប្រកាសតែងតែលើសពីអ្វីដែលមនុស្សដូចគ្នានឹងបង់ នៅពេលមានការជួញដូរថវិកាពិតប្រាកដ គឺភាពលំអៀងសម្មតិកម្ម។
- **មិនអើពើឥទ្ធិពលបង្កប់។** ទំនិញដូចគ្នាដែលវាយតម្លៃតែឯង ធៀបនឹងវាយតម្លៃជាផ្នែកនៃកញ្ចប់ធំជាង ផលិតការប៉ាន់ស្មាន WTP ខុសគ្នា ត្រូវរាយការណ៍ថាមានអ្វីផ្សេងទៀត ប្រសិនបើមាន នៅក្នុងក្របខ័ណ្ឌស្ទង់មតិ។
- **ការចាត់ទុកការប៉ាន់ស្មានចំណុចរបស់ការស្ទង់មតិតែមួយថាបានសម្រេច។** ការអនុវត្តរបស់ Green Book រំពឹងចន្លោះមួយ និងការពិភាក្សាអំពីភាពលំអៀងដែលស្គាល់ មិនមែនលេខទទេដែលដាក់បន្តទៅក្នុងតារាងថ្លៃដើម-អត្ថប្រយោជន៍ដូចជាវាជាតម្លៃទីផ្សារទេ។
