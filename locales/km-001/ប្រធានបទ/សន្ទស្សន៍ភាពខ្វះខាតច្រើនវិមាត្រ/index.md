# សន្ទស្សន៍ភាពខ្វះខាតច្រើនវិមាត្រ (IMD)

IMD គឺជារង្វាស់ផ្លូវការនៃភាពខ្វះខាតដែលទាក់ទងសម្រាប់តំបន់តូចៗក្នុងប្រទេសអង់គ្លេស ដោយចាត់ថ្នាក់ Lower-layer Super Output Areas (LSOA) ទាំង 32,844 របស់ប្រទេស (នីមួយៗមានអ្នករស់នៅប្រហែល 1,500 នាក់) ពី 1 (ខ្វះខាតបំផុត) ដល់ 32,844 (ខ្វះខាតតិចបំផុត)។ វាត្រូវបានបោះពុម្ពផ្សាយដោយអ្វីដែលឥឡូវនេះជា Ministry of Housing, Communities and Local Government (MHCLG ពីមុន MHCLG/DCLG) ចុងក្រោយជា English Indices of Deprivation 2019 ហើយវាកំណត់ផ្លូវដោយផ្ទាល់នូវមូលនិធិរដ្ឋាភិបាលកណ្តាល ការកំណត់អាទិភាពសុខភាពសាធារណៈ និងសិទ្ធិទទួលសម្រាប់គម្រោងមូលដ្ឋានរាប់សិប។

## ហេតុអ្វីបានជាសំខាន់

ភាពខ្វះខាតមិនមែនជារឿងតែមួយទេ — សង្កាត់មួយអាចក្រីក្រផ្នែកប្រាក់ចំណូលប៉ុន្តែមានសុវត្ថិភាព ឬមានប្រាក់ចំណូលគ្រប់គ្រាន់ប៉ុន្តែរងទុក្ខពីលទ្ធផលសុខភាពអន់ និងលំនៅដ្ឋានអន់។ សន្ទស្សន៍មុនរបស់ IMD (ដែលមានតាំងពីសូចនាករភាពខ្វះខាតរបស់ Department of the Environment ក្នុងទសវត្សរ៍ 1970) បានវិវត្តទៅជាគំរូប្រាំពីរវិស័យសព្វថ្ងៃ ពិតជាព្រោះការកំណត់គោលដៅដោយសូចនាករតែមួយ (ឧទាហរណ៍ អត្រាគ្មានការងារតែម្យ៉ាង) ច្រើនតែខកខានតំបន់ដែលខ្វះខាតតាមផ្លូវផ្សេង។ IMD 2019 រួមបញ្ចូលប្រាក់ចំណូល ការងារ ការអប់រំ សុខភាព ឧក្រិដ្ឋកម្ម ឧបសគ្គចំពោះលំនៅដ្ឋាន និងសេវា និងបរិយាកាសរស់នៅ ទៅជាចំណាត់ថ្នាក់សមាសមួយក្នុង LSOA នីមួយៗ វិស័យនីមួយៗសង់ពីកញ្ចប់សូចនាករផ្ទាល់ខ្លួន ហើយថ្លឹងថ្លែងតាមវិធីសាស្ត្ររបស់ MHCLG។ ពីព្រោះវាដំណើរការនៅកម្រិតតំបន់តូច (LSOA) ជាជាងកម្រិតអាជ្ញាធរមូលដ្ឋាន វាបង្ហាញលំហូរភាពខ្វះខាតដែលលាក់ក្នុងស្រុកដែលមានបានផ្សេងទៀត — ហេតុផលដែល IMD មិនមែនប្រាក់ចំណូលមធ្យមរបស់អាជ្ញាធរមូលដ្ឋានទេ ដែល NHS England, pupil premium របស់ Department for Education និងរូបមន្តមូលនិធិរបស់អាជ្ញាធរមូលដ្ឋានរាប់សិបពិតជាផ្អែកលើ។ កម្មវិធីដែលកំណត់សិទ្ធិទទួល កំណត់អាទិភាពការចុះជួយ ឬរាយការណ៍ផលប៉ះពាល់តាមតំបន់ក្នុងប្រទេសអង់គ្លេស គួរចាត់ទុកដេស៊ីល ឬចំណាត់ថ្នាក់ IMD ជាធាតុចូលលំដាប់ទីមួយ មិនមែនការគិតបន្ថែមទេ — ហើយនៅកន្លែងដែលកម្មវិធីកំណត់គោលដៅតំបន់ខ្វះខាតបំផុតដោយចេតនា ការវាយតម្លៃរបស់វាគួរអនុវត្ត [ការថ្លឹងថ្លែងតាមការបែងចែក](../ការថ្លឹងថ្លែងតាមការបែងចែក/) ស្របនឹងការកំណត់គោលដៅនោះ ជាជាងវាយតម្លៃផោនអត្ថប្រយោជន៍ដូចគ្នាមិនថាវាធ្លាក់កន្លែងណា។

## គណិតវិទ្យា

```
7 domains, weighted:
  Income                              22.5%
  Employment                          22.5%
  Education, Skills and Training      13.5%
  Health Deprivation and Disability   13.5%
  Crime                                9.3%
  Barriers to Housing and Services     9.3%
  Living Environment                   9.3%

Each domain score: indicators standardized (ranked, then transformed
towards a normal distribution) and combined by exponential transformation
so that high deprivation on any one indicator cannot be fully cancelled
out by low deprivation on others within that domain.

IMD composite score (LSOA) = Σ (domain score × domain weight)
Rank LSOAs by composite score → 1 (most deprived) to 32,844 (least deprived)
Deciles: rank ÷ 3,284 (approx.), decile 1 = most deprived 10% of LSOAs
```

## ឧទាហរណ៍គណនា

**ពិន្ទុសមាស LSOA** ដោយប្រើពិន្ទុវិស័យស្តង់ដារបង្ហាញឧទាហរណ៍ (0 = គ្មានសញ្ញាភាពខ្វះខាត ខ្ពស់ជាង = ខ្វះខាតជាង)៖

```
Income               0.35 × 0.225 = 0.07875
Employment           0.30 × 0.225 = 0.06750
Education            0.20 × 0.135 = 0.02700
Health               0.15 × 0.135 = 0.02025
Crime                0.10 × 0.093 = 0.00930
Barriers to Housing  0.05 × 0.093 = 0.00465
Living Environment   0.08 × 0.093 = 0.00744

Composite score = 0.07875 + 0.06750 + 0.02700 + 0.02025
                + 0.00930 + 0.00465 + 0.00744  = 0.21489
```

បន្ទាប់មកពិន្ទុសមាសនោះត្រូវបានចាត់ថ្នាក់ធៀបនឹងពិន្ទុរបស់ LSOA ទាំង 32,844។ ប្រសិនបើវាដាក់ LSOA នៅចំណាត់ថ្នាក់ 2,950 វាធ្លាក់ក្នុងដេស៊ីល 1 (2,950 ÷ 3,284 ≈ 0.9 មានន័យថាក្នុង 10% ខ្វះខាតបំផុតនៃសង្កាត់ក្នុងប្រទេសអង់គ្លេស) — ដែលសម្រាប់រូបមន្តមូលនិធិជាច្រើនគឺជាកម្រិតដែលបើកសិទ្ធិទទួល មិនថាអាជ្ញាធរមូលដ្ឋានជុំវិញទទួលបានពិន្ទុជាមធ្យមប៉ុន្មានទេ។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

- សេវាណាដែលកំណត់ទីតាំងភូមិសាស្ត្រអ្នកប្រើប្រាស់តាមកូដប្រៃសណីយ៍ ឬ LSOA អាចភ្ជាប់តារាងរកមើល IMD ដែលបានបោះពុម្ពផ្សាយ (CSV ឥតគិតថ្លៃ មានកំណែពី MHCLG) ដើម្បីបន្ថែមដេស៊ីលភាពខ្វះខាតជា covariate — សម្រាប់កំណត់គោលដៅការចុះជួយ កំណត់អាទិភាពបន្ទុកករណី ឬរាយការណ៍លទ្ធផលតាមក្រុមភាពខ្វះខាត ដោយមិនប្រមូលទិន្នន័យផ្ទាល់ខ្លួនថ្មី។
- ដេស៊ីល IMD គឺជាការត្រួតពិនិត្យសមធម៌ស្តង់ដារសម្រាប់សេវាឌីជីថលសាធារណៈ៖ ការបូកតារាងឆ្លងការទទួលយកសេវា ការបាត់បង់ ឬការពេញចិត្តតាមដេស៊ីល IMD លេចចេញគម្លាតការចូលដំណើរការដែលរង្វាស់រួមលាក់ — សូមមើល [ការរួមបញ្ចូលឌីជីថល](../ការរួមបញ្ចូលឌីជីថល/) និង [រង្វាស់ការពេញចិត្តរបស់ពលរដ្ឋ](../រង្វាស់ការពេញចិត្តរបស់ពលរដ្ឋ/)។
- ពីព្រោះចំណាត់ថ្នាក់ IMD គឺទាក់ទង (វាតែងតែបូកទៅសំណុំចំណាត់ថ្នាក់ថេរទូទាំងប្រទេសអង់គ្លេស) វាមិនអាចបង្ហាញថាតើភាពខ្វះខាតជាតិកំពុងកើនឡើង ឬថយចុះតាមពេលវេលាទេ — គ្រាន់តែថាតំបន់ណាចាត់ថ្នាក់នៅកន្លែងណាធៀបនឹងគ្នាក្នុងការបោះពុម្ពនោះ។ កុំសាងសង់ផ្ទាំងគ្រប់គ្រងនិន្នាការដាច់ខាតលើចំណាត់ថ្នាក់ IMD ឆៅតែម្យ៉ាង។

## កំហុសដែលតែងកើតមាន

- **ប្រៀបធៀបចំណាត់ថ្នាក់ IMD ឆ្លងការបោះពុម្ព (2015 ធៀប 2019) ជានិន្នាការតាមពេលវេលា** — សូចនាករមូលដ្ឋាន ភូមិសាស្ត្រ និងវិធីសាស្ត្រទាំងអស់ផ្លាស់ប្តូររវាងការបោះពុម្ព។ MHCLG ណែនាំច្បាស់លាស់ឱ្យកុំប្រើការផ្លាស់ប្តូរចំណាត់ថ្នាក់ជាភស្តុតាងថាតំបន់មួយកាន់តែខ្វះខាត ឬតិចជាង។
- **អនុវត្ត IMD កម្រិត LSOA ទៅបុគ្គល** — LSOA ក្នុងដេស៊ីល 1 នៅតែមានគ្រួសារមិនខ្វះខាត ហើយ LSOA ដេស៊ីល 10 នៅតែមានគ្រួសារខ្វះខាត។ IMD ពណ៌នាតំបន់ មិនមែនមនុស្សទេ ហើយការប្រើវាជាតំណាងសិទ្ធិទទួលបុគ្គលចាត់ថ្នាក់ខុសទាំងពីរទិស។
- **មិនអើពើព័ត៌មានលម្អិតកម្រិតវិស័យ ដោយពេញចិត្តចំណាត់ថ្នាក់សមាស** — LSOA ពីរដែលមានពិន្ទុសមាសដូចគ្នាអាចមានទម្រង់វិស័យខុសគ្នាទាំងស្រុង (មួយខ្វះខាតសុខភាព មួយខ្វះខាតឧក្រិដ្ឋកម្ម)។ គម្រោងកំណត់គោលដៅដែលមានគោលបំណងចំពោះបញ្ហាមួយ គួរប្រើពិន្ទុវិស័យពាក់ព័ន្ធ មិនមែនសមាសលាយទេ។
