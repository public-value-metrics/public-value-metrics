# តម្លៃនៃបញ្ញាសិប្បនិមិត្តក្នុងរដ្ឋាភិបាល

តម្លៃនៃបញ្ញាសិប្បនិមិត្ត (AI) ក្នុងរដ្ឋាភិបាល គឺជាតម្រូវការដែលថា ប្រព័ន្ធ AI ដែលប្រើក្នុងសេវាសាធារណៈ ត្រូវឆ្លងកាត់ស្តង់ដារតម្លៃទទួលបានធៀបនឹងថ្លៃចំណាយ (value for money) និងតម្លៃសាធារណៈដូចការសម្រេចចិត្តចំណាយផ្សេងទៀតដែរ មិនមែនទាបជាងដោយសារវាថ្មី ហើយក៏មិនមែនខ្ពស់ជាងដោយសារតែការភ័យខ្លាចនោះដែរ។ វាជាសំណួរដែលក្រុមអនុវត្តត្រូវអាចឆ្លើយបានមុន មិនមែនក្រោយពេលមុខងារ AI ត្រូវបានដាក់ឱ្យប្រើ៖ តើវាបង្កើតតម្លៃច្រើនជាងថ្លៃចំណាយដែរឬទេ នៅពេលបានគិតថ្លៃនៃការធានាគុណភាព ការត្រួតពិនិត្យ និងហានិភ័យដោយស្មោះត្រង់?

## ហេតុអ្វីបានជាសំខាន់

ការិយាល័យឌីជីថល និងទិន្នន័យកណ្តាលរបស់ចក្រភពអង់គ្លេស (Central Digital and Data Office ឬ CDDO) បានបោះពុម្ពផ្សាយ Generative AI Framework for Government ក្នុងឆ្នាំ 2024 ដោយផ្អែកលើការណែនាំបណ្តោះអាសន្នមុននេះកាលពីខែមិថុនា ឆ្នាំ 2023 ហើយរៀបចំវាជុំវិញគោលការណ៍ដប់ ដែលគ្របដណ្តប់លើអ្វីដែល AI បង្កើតខ្លឹមសារ (generative AI) ជា ផលប៉ះពាល់ខាងសីលធម៌ សុវត្ថិភាពឧបករណ៍ ការត្រួតពិនិត្យធានាគុណភាព ការគ្រប់គ្រងវដ្តជីវិតពេញលេញរបស់ generative AI ការកំណត់ករណីប្រើប្រាស់ពិតប្រាកដ ការសហការទូទាំងរដ្ឋាភិបាល តម្លាភាព ជំនាញ និងអភិបាលកិច្ច។ ការបញ្ជាក់ខ្ជាប់ខ្ជួនរបស់ក្របខណ្ឌនេះអំពី «ការគ្រប់គ្រងដោយមនុស្សប្រកបដោយអត្ថន័យ» និងការគ្រប់គ្រងវដ្តជីវិតពេញលេញ កើតឡើងព្រោះករណីអាជីវកម្មគម្រោង AI មានទម្រង់បរាជ័យជាក់លាក់មួយ ដែលការចំណាយលើ IT ផ្សេងទៀតមិនមាន៖ តួលេខផលិតភាពចំណងជើងនៃគម្រោងសាកល្បងងាយបង្កើត និងងាយបំផ្លើស ព្រោះវាត្រូវបានវាស់មុនពេលគិតដល់បន្ទុកនៃការផ្ទៀងផ្ទាត់ ការកែតម្រូវ និងការត្រួតពិនិត្យដែលឧបករណ៍នេះបង្កើត។ ក្រៅពីក្របខណ្ឌនេះ ស្តង់ដារកត់ត្រាតម្លាភាពក្បួនដោះស្រាយ (Algorithmic Transparency Recording Standard ឬ ATRS) តម្រូវឱ្យស្ថាប័នសាធារណៈបោះពុម្ពផ្សាយកំណត់ត្រាស្តង់ដារ គឺគោលបំណង ទិន្នន័យដែលប្រើ ប្រសិទ្ធភាព ការសាកល្បងភាពយុត្តិធម៌ និងការរៀបចំត្រួតពិនិត្យដោយមនុស្ស សម្រាប់ឧបករណ៍ក្បួនដោះស្រាយដែលមានឥទ្ធិពលសំខាន់លើការសម្រេចចិត្តអំពីបុគ្គល ដែលធ្វើឱ្យថ្លៃធានាគុណភាពនៃប្រព័ន្ធ AI ក្លាយជារឿងកំណត់ត្រាសាធារណៈ មិនមែនជាការប៉ាន់ស្មានផ្ទៃក្នុងដែលក្រុមអាចរំលងដោយស្ងាត់ស្ងៀមបានឡើយ។

## គណិតវិទ្យា

ការទទួលយក AI ត្រូវបានវាយតម្លៃជាការបន្ថែមលើ មិនមែនជាការជំនួស ការវាយតម្លៃ [តម្លៃទទួលបានធៀបនឹងថ្លៃចំណាយ](../តម្លៃសមស្របនឹងថ្លៃចំណាយ/) ស្តង់ដារទេ ដោយធ្វើឱ្យលក្ខខណ្ឌជាក់លាក់របស់ AI ច្បាស់លាស់ ជាជាងបញ្ចូលវាទៅក្នុងតួលេខ «កំណើនផលិតភាព» តែមួយ៖

```
Net value of an AI system =
    productivity gain (time saved × loaded staff cost)
  − licence/compute cost
  − human verification and oversight cost (checking AI output before
    it's acted on — this does not shrink to zero even for mature tools)
  − ATRS documentation and ongoing monitoring cost
  − risk-adjusted cost of harm from errors, bias, or hallucination,
    weighted by who bears that harm (distributional-weighting)

A pilot productivity figure that omits the oversight term is not
comparable to a business-as-usual cost baseline that already includes
equivalent human review — see ai-productivity-in-the-public-sector
for the fuller productivity-measurement discipline this borrows from.
```

## ឧទាហរណ៍គណនា

**អាជ្ញាធរមូលដ្ឋានប្រើឧបករណ៍ generative AI ដើម្បីព្រាងការឆ្លើយតបដំបូងចំពោះសំណួរប្រចាំអំពីពន្ធក្រុមប្រឹក្សា (council tax)**៖ 25,000 សំណួរ/ឆ្នាំ ពីមុនត្រូវបានដោះស្រាយទាំងស្រុងដោយអ្នកចាត់ការករណី ជាមធ្យម 14 នាទី/សំណួរ ថ្លៃចំណាយបុគ្គលិកពេញលេញ £34/ម៉ោង។

```
Baseline (no AI) cost:
  25,000 × (14/60) × £34 = £198,333/year

Pilot headline claim: AI drafts a response in 90 seconds,
caseworker "just reviews and sends" — claimed new time is 3 minutes
  25,000 × (3/60) × £34 = £42,500/year
  → claimed saving £155,833/year (looks transformational)

Fully-loaded figure, measured after 3 months live rather than in the
pilot's hand-picked test cases:
  Actual review + correction time per response: 6 minutes (drafts
  need real editing for complex or emotionally sensitive enquiries)
  25,000 × (6/60) × £34 = £85,000/year
  Licence/compute cost: £38,000/year
  ATRS documentation and quarterly bias/quality monitoring: £14,000/year
  Total cost = 85,000 + 38,000 + 14,000 = £137,000/year

Real saving = 198,333 − 137,000 = £61,333/year — genuine and worth
keeping, but well under half the pilot's headline claim, and it
required an honest oversight-time measurement, not the pilot's
best-case one, to find.
```

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

នេះគឺជាចំណុចជួបគ្នារវាង [ផលិតភាព AI ក្នុងវិស័យសាធារណៈ](../ផលិតភាព-ai-ក្នុងវិស័យសាធារណៈ/) និងប្រធានបទនេះ៖ ក្រុមវិស្វកម្មដែលបង្កើតមុខងារ AI ក្នុងសេវាសាធារណៈ ជាម្ចាស់ឧបករណ៍វាស់វែងដែលធ្វើឱ្យតួលេខ «ពិត» ក្នុងឧទាហរណ៍អនុវត្តអាចកើតមាន គឺការកត់ត្រាពេលវេលាពិនិត្យជាក់ស្តែង គម្លាតកែប្រែរវាងសេចក្តីព្រាង និងការឆ្លើយតបដែលបានផ្ញើ និងអត្រាបញ្ជូនបន្ត ជាជាងជឿលើលក្ខខណ្ឌបង្ហាញរបស់គម្រោងសាកល្បង។ មុខងារ AI គួរត្រូវបានវាយតម្លៃធៀបនឹងចំណុចទី 9 នៃ [ស្តង់ដារសេវាឌីជីថល](../ស្តង់ដារសេវាឌីជីថល/) (សេវាមានសុវត្ថិភាព ភាពឯកជនរបស់អ្នកប្រើ) ហើយយោងចំណាំជាមួយ [តម្លៃសន្តិសុខអ៊ីនធឺណិតក្នុងវិស័យសាធារណៈ](../តម្លៃនៃសន្តិសុខអ៊ីនធឺណិតក្នុងវិស័យសាធារណៈ/) នៅពេលឧបករណ៍ប៉ះពាល់ទិន្នន័យពលរដ្ឋ ហើយប្រព័ន្ធ AI ណាដែលមានឥទ្ធិពលសំខាន់លើការសម្រេចចិត្តអំពីបុគ្គល ត្រូវការកំណត់ត្រា ATRS មុននឹងអាចចាត់ទុកថាត្រៀមខ្លួនសម្រាប់ការវាយតម្លៃ ដូចគ្នានឹងសេវាមួយត្រូវការឆ្លងកាត់ការវាយតម្លៃ [ស្តង់ដារសេវាឌីជីថល](../ស្តង់ដារសេវាឌីជីថល/) មុនដាក់ឱ្យប្រើដែរ។

## កំហុសដែលតែងកើតមាន

- **AI-washing**៖ ការដាក់ស្លាកស្វ័យប្រវត្តិកម្មផ្អែកលើច្បាប់ដែលមានស្រាប់ថាជា «AI» ដើម្បីទទួលបានមូលនិធិ ឬការចាប់អារម្មណ៍ដែលបម្រុងទុកសម្រាប់ការទទួលយក AI ដោយគ្មានហានិភ័យភាពត្រឹមត្រូវ ឬភាពលំអៀងដែលពិតជាបញ្ជាក់ពីការត្រួតពិនិត្យបន្ថែមរបស់ក្របខណ្ឌ។
- **ការវាស់ផលិតភាពគម្រោងសាកល្បង មិនមែនផលិតភាពផលិតកម្មពិត**៖ គម្រោងសាកល្បងដំណើរការលើករណីសាកល្បងដែលបានជ្រើសរើស ជាមួយអ្នកពិនិត្យដែលយកចិត្តទុកដាក់ ចំណែកផលិតកម្មពិតដំណើរការលើករណីចម្រុះពេញលេញ ជាមួយអ្នកពិនិត្យដែលតាមពេលវេលាបង្កើតភាពលំអៀងស្វ័យប្រវត្តិកម្ម (automation bias) ហើយពិនិត្យលទ្ធផលមិនគ្រប់គ្រាន់ ហើយទាំងពីរនេះធ្វើឱ្យតួលេខថ្លៃត្រួតពិនិត្យដោយស្មោះត្រង់ខុសប្រក្រតី។
- **ការរំលងការចុះបញ្ជី ATRS ដោយសារឧបករណ៍ «មិនមែនជាការសម្រេចចិត្តស្វ័យប្រវត្តិពិតប្រាកដ»**៖ កម្រិតរបស់ស្តង់ដារគឺឥទ្ធិពលសំខាន់លើការសម្រេចចិត្តអំពីបុគ្គល ដែលឧបករណ៍ព្រាង ឬចាត់ថ្នាក់ AI ប្រឈមមុខនឹងពលរដ្ឋភាគច្រើនបំពេញបាន ទោះបីមនុស្សចុះហត្ថលេខាជាផ្លូវការក៏ដោយ។
- **ការមិនអើពើនឹងផលប៉ះពាល់ចែកចាយនៃកំហុស**៖ អត្រាកំហុសជាមធ្យមរបស់ប្រព័ន្ធ AI លើអ្នកប្រើទាំងអស់ អាចលាក់អត្រាកំហុស ឬភាពលំអៀងខ្ពស់ជាងសម្រាប់ក្រុមជាក់លាក់ ដូច្នេះ [ការថ្លឹងថ្លែងតាមការចែកចាយ](../ការថ្លឹងថ្លែងតាមការបែងចែក/) គួរត្រូវបានអនុវត្តលើលក្ខខណ្ឌគ្រោះថ្នាក់ដែលបានកែតម្រូវហានិភ័យ មិនមែនត្រឹមតែតួលេខភាពត្រឹមត្រូវសរុបប៉ុណ្ណោះទេ។
