# ឆ្នាំជីវិតកែតម្រូវតាមសុខុមាលភាព (WELLBY)

WELLBY គឺជាចំណុចបន្ថែមមួយនៃការពេញចិត្តក្នុងជីវិត នៅលើមាត្រដ្ឋានសុខុមាលភាពស្តង់ដារ 0–10 សម្រាប់មនុស្សម្នាក់ក្នុងរយៈពេលមួយឆ្នាំ។ វាជាឯកតាស្រដៀង QALY ដែលប្រើក្នុងសេដ្ឋកិច្ចសុខភាព ជាឯកតាតែមួយដែលអនុញ្ញាតឱ្យអ្នកប្រៀបធៀបអន្តរាគមន៍ដែលលទ្ធផលរបស់វាមិនមានអ្វីរួមគ្នា ប៉ុន្តែកសាងលើសុខុមាលភាពប្រធានបទ ជាជាងស្ថានភាពសុខភាពគ្លីនិក ហើយត្រូវបានចែងក្នុង "Wellbeing guidance for appraisal: supplementary Green Book guidance" (2021) របស់ក្រសួងរតនាគារអង់គ្លេស (HM Treasury)។

## ហេតុអ្វីបានជាសំខាន់

ការវាយតម្លៃថ្លៃដើម-អត្ថប្រយោជន៍ត្រូវការឯកតារួម ដើម្បីប្រៀបធៀបជំនួយក្លឹបយុវជន ធៀបនឹងគម្រោងសុវត្ថិភាពផ្លូវថ្នល់ ធៀបនឹងសេវាសុខភាពផ្លូវចិត្ត ដែលគ្មានមួយណាចែករំលែកវិធានការលទ្ធផលរួម។ សេដ្ឋកិច្ចសុខភាពបានដោះស្រាយបញ្ហានេះសម្រាប់អន្តរាគមន៍គ្លីនិកដោយ QALY គឺឆ្នាំជីវិតកែតម្រូវតាមគុណភាព ដែលថ្លឹងពី 0 (ស្លាប់) ដល់ 1 (សុខភាពពេញលេញ)។ ការណែនាំសុខុមាលភាពរបស់ HM Treasury ពង្រីកតក្កវិជ្ជាដូចគ្នាទៅការចំណាយសាធារណៈមិនមែនសុខភាព ដោយប្រើសំណួរការពេញចិត្តក្នុងជីវិតដែលស្របគ្នារបស់ ONS ("Overall, how satisfied are you with your life nowadays?" ឆ្លើយ 0–10) ជាជណ្តើរលទ្ធផល ជំនួសឱ្យសន្ទស្សន៍ស្ថានភាពសុខភាព។ WELLBY 1 មានន័យថាការពេញចិត្តក្នុងជីវិតរបស់មនុស្សម្នាក់កើនឡើងមួយពិន្ទុពេញក្នុងរយៈពេលមួយឆ្នាំ (ឬស្មើគ្នា ការពេញចិត្តរបស់មនុស្សដប់នាក់កើនឡើង 0.1 ពិន្ទុម្នាក់ៗក្នុងមួយឆ្នាំ WELLBY បូកសរុបឆ្លងកាត់ប្រជាជនដូច QALY ដែរ)។ ការណែនាំរបស់ HM Treasury កំណត់តម្លៃជាប្រាក់ឧទាហរណ៍ក្នុងមួយ WELLBY (ប្រហែល £13,000 តម្លៃឆ្នាំ 2019/20) ដែលបានមកពីការផ្សះផ្សាទិន្នន័យសុខុមាលភាពប្រធានបទ ជាមួយវិធីសាស្ត្រផ្សេងទៀតចំពោះតម្លៃនៃឆ្នាំជីវិត ផ្តល់ឱ្យអ្នកវាយតម្លៃនូវមធ្យោបាយបម្លែងលទ្ធផលជាប្រាក់ គឺការកាត់បន្ថយភាពឯកកោ សាមគ្គីភាពសហគមន៍ ការចូលដល់ទីធ្លាបៃតង ដែលបច្ចេកទេស[ការវាយតម្លៃសុខុមាលភាព](../ការវាយតម្លៃសុខុមាលភាព/) ពីមុនអាចគ្រាន់តែពិពណ៌នា មិនអាចប្រៀបធៀបលើមូលដ្ឋានរួមជាមួយការចំណាយសុខភាព ឬសុវត្ថិភាពបានទេ។

## គណិតវិទ្យា

```
WELLBY = Δ ការពេញចិត្តក្នុងជីវិត (មាត្រដ្ឋាន 0–10) × ចំនួនឆ្នាំដែលការផ្លាស់ប្តូរនៅស្ថិតស្ថេរ
        (បូកសរុបលើមនុស្សទាំងអស់ដែលរងផលប៉ះពាល់)

អត្ថប្រយោជន៍សុខុមាលភាពជាប្រាក់ = WELLBY ដែលបានបង្កើត × តម្លៃក្នុងមួយ WELLBY (តម្លៃយោង HMT)

ប្រៀបធៀប QALY = Δ ឧបយោគស្ថានភាពសុខភាព (មាត្រដ្ឋាន 0–1) × ឆ្នាំដែលរស់នៅក្នុងស្ថានភាពនោះ
```

មាត្រដ្ឋានការពេញចិត្ត 0–10 និងមាត្រដ្ឋានឧបយោគ QALY 0–1 មិនអាចប្តូរគ្នាបានទេ ដោយគ្មានជំហានបម្លែង ការណែនាំរបស់ HM Treasury ពិភាក្សាការផ្សះផ្សាទាំងពីរ ដើម្បីឱ្យឧទាហរណ៍ អន្តរាគមន៍សុខភាពដែលបានវាយតម្លៃជា QALY និងអន្តរាគមន៍សង្គមដែលបានវាយតម្លៃជា WELLBY មិនត្រូវបានរាប់ស្ទួនដោយស្ងាត់ស្ងៀម ឬទុកឱ្យមិនអាចប្រៀបធៀបបានក្នុង[ការវាយតម្លៃតាម Green Book](../ការវាយតម្លៃតាម-green-book/) តែមួយ។

## ឧទាហរណ៍គណនា

**សេវាភាពឯកកោរបស់អាជ្ញាធរមូលដ្ឋាន**៖ គម្រោងមិត្តភាពបម្រើអ្នករស់នៅចាស់ឯកកោ 400 នាក់។ ការស្ទង់មតិតាមដានបង្ហាញថាការពេញចិត្តក្នុងជីវិតមធ្យមកើនពី 5.2 ទៅ 6.0 (កើន 0.8 ពិន្ទុ) ហើយឥទ្ធិពលត្រូវបានប៉ាន់ស្មានថាស្ថិតស្ថេរ 2 ឆ្នាំមុនរសាត់។

```
WELLBY = 400 នាក់ × 0.8 ពិន្ទុ × 2 ឆ្នាំ = 640 WELLBY

តម្លៃជាប្រាក់ = 640 × £13,000 = £8,320,000
```

ធៀបនឹងថ្លៃចំណាយកម្មវិធីប្រចាំឆ្នាំ £300,000 (£600,000 ក្នុង 2 ឆ្នាំ) សមាមាត្រអត្ថប្រយោជន៍-ថ្លៃចំណាយប្រហែល 8,320,000 / 600,000 ≈ **13.9:1** ជាតួលេខដែលឥឡូវនេះអាចស្ថិតក្នុងតារាងវាយតម្លៃដូចគ្នានឹងថ្លៃចំណាយក្នុងមួយ QALY ដែលបានជៀសវាងរបស់គម្រោងសុខភាព ឬការសន្សំពេលវេលាធ្វើដំណើររបស់គម្រោងដឹកជញ្ជូន។

**អង្គការសប្បុរសធម៌ មាត្រដ្ឋានតូចជាង**៖ កម្មវិធីសិល្បៈសហគមន៍ទៅដល់អ្នកចូលរួម 50 នាក់ ដែលមានការកើនការពេញចិត្តដែលបានវាស់ 0.3 ពិន្ទុ មានរយៈពេល 1 ឆ្នាំ។

```
WELLBY = 50 × 0.3 × 1 = 15 WELLBY
តម្លៃជាប្រាក់ = 15 × £13,000 = £195,000
```

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

- សេវាប្រឈមមុខនឹងពលរដ្ឋណាមួយដែលប្រមូលធាតុស្ទង់មតិការពេញចិត្តក្នុងជីវិត ឬសុខុមាលភាពរួចហើយ (វេទិកាអាជ្ញាធរមូលដ្ឋាន និងសុខភាព និងថែទាំជាច្រើនធ្វើ តាមសំណួរសុខុមាលភាពស្តង់ដារបួនរបស់ ONS) អាចគណនា WELLBY ដោយផ្ទាល់ពីបំពង់ទិន្នន័យដែលមានស្រាប់ ជាជាងចាត់ចែងការវាយតម្លៃសេដ្ឋកិច្ចជាក់លាក់សម្រាប់រាល់ការផ្លាស់ប្តូរសេវា។
- WELLBY ផ្តល់ក្រុមវិស្វកម្មដែលកសាងសម្រាប់ការរាយការណ៍[ច្បាប់តម្លៃសង្គម](../ច្បាប់តម្លៃសង្គម/) ឬ
  [ផលចំណេញសង្គមលើការវិនិយោគ](../ផលចំណេញសង្គមលើការវិនិយោគ/) នូវភាគបែងស្តង់ដារជាតិ ដែលអនុម័តដោយ HM Treasury ជៀសវាងការកើនឡើងនៃ "ពិន្ទុផលប៉ះពាល់" ជាក់លាក់ដែលមិនអាចប្រៀបធៀបឆ្លងកិច្ចសន្យា ឬអ្នកផ្គត់ផ្គង់។
- ដោយសារ WELLBY អាចបូកបានឆ្លងមនុស្ស និងពេលវេលា ពួកវាបង្កើតបានស្អាតក្នុងការតាមដានលទ្ធផលកម្រិតប្រជាជន ដែលប្រើក្នុងប្រព័ន្ធ[គណនេយ្យភាពផ្អែកលើលទ្ធផល](../គណនេយ្យភាពផ្អែកលើលទ្ធផល/) ផ្ទាំងគ្រប់គ្រងសេវាអាចរាយការណ៍ WELLBY បង្គរដែលបានបង្កើតក្នុងមួយត្រីមាស ដូចដែលប្រព័ន្ធសុខភាពរាយការណ៍ QALY ដែលទទួលបាន។

## កំហុសដែលតែងកើតមាន

- **ការសន្មត់ថាការកើនការពេញចិត្តដែលរាយការណ៍ដោយខ្លួនឯង អាចសន្មត់ទាំងស្រុងទៅអន្តរាគមន៍** បើគ្មានcounterfactual (ក្រុមប្រៀបធៀប ឬការរចនាមុន/ក្រោយមានការគ្រប់គ្រង) អ្នកមិនអាចញែក WELLBY ដែលកើនចេញពីនិន្នាការទូទៅបានទេ សូមមើល [ការវិភាគ counterfactual](../ការវិភាគ-counterfactual/)។
- **ការលាយ WELLBY និង QALY ក្នុងសរុបតែមួយដោយមិនផ្សះផ្សា** ការណែនាំរបស់ HM Treasury ចែងច្បាស់ថា ទាំងពីរប្រើមាត្រដ្ឋានខុសគ្នា និងទ្រឹស្តីតម្លៃមូលដ្ឋានខុសគ្នា ការបូកវាដោយឆោតល្ងង់ រាប់ស្ទួនសុខុមាលភាពដែលត្រួតគ្នា។
- **ការប្រើតម្លៃជាប្រាក់យោងដោយមិនពិនិត្យ** តួលេខ £ ក្នុងមួយ WELLBY គឺជាការប៉ាន់ស្មានមធ្យមភាគជាតិដែលមានចន្លោះមិនប្រាកដប្រជាពិតប្រាកដ ការណែនាំរបស់ HM Treasury ណែនាំការវិភាគភាពរសើប មិនមែនចាត់ទុកវាជាអត្រាប្តូរថេរទេ។
